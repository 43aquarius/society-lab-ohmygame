     1→/**
     2→ * 建筑渲染 — 程序化等距建筑
     3→ * 每种建筑有独特剪影与时代色板；建造中的脚手架；
     4→ * 农田作物随年度周期生长。
     5→ */
     6→import { BuildingKind, type Building, type WorldState } from '../types';
     7→import { TILE_W, TILE_H, ELEV_H, vhash } from './iso';
     8→
     9→export interface BuildingPalette {
    10→  wood: string; woodDark: string; roof: string; roofDark: string;
    11→  stone: string; stoneDark: string; accent: string; cloth: string;
    12→}
    13→
    14→const P = (wood: string, roof: string, stone = 'rgb(168,162,152)', accent = 'rgb(194,120,60)', cloth = 'rgb(224,200,150)'): BuildingPalette => ({
    15→  wood, woodDark: shade(wood, 0.72), roof, roofDark: shade(roof, 0.7),
    16→  stone, stoneDark: shade(stone, 0.72), accent, cloth,
    17→});
    18→
    19→function shade(rgb: string, f: number): string {
    20→  const m = rgb.match(/rgb\((\d+),(\d+),(\d+)\)/);
    21→  if (!m) return rgb;
    22→  return `rgb(${(+m[1]) * f | 0},${(+m[2]) * f | 0},${(+m[3]) * f | 0})`;
    23→}
    24→
    25→export const PALETTES: Record<number, BuildingPalette> = {
    26→  0: P('rgb(148,116,80)', 'rgb(201,162,94)'),          // 原始：茅草
    27→  1: P('rgb(140,108,72)', 'rgb(178,120,70)'),          // 农业：木瓦
    28→  2: P('rgb(132,102,70)', 'rgb(156,86,60)', 'rgb(178,172,160)'), // 城镇：陶瓦
    29→  3: P('rgb(120,96,84)', 'rgb(88,88,100)', 'rgb(150,146,140)', 'rgb(80,96,120)'), // 工业：砖与铁
    30→  4: P('rgb(150,140,128)', 'rgb(90,110,116)', 'rgb(196,192,184)', 'rgb(47,126,118)'), // 现代：水泥与青
    31→};

    33→/** 建筑占地面积（格）：多数 1 格，大型 2 格 */
    34→function footprint(kind: BuildingKind): { w: number; d: number } {
    35→  switch (kind) {
    36→    case BuildingKind.Admin: return { w: 1.6, d: 1.4 };
    37→    case BuildingKind.Factory: return { w: 1.8, d: 1.6 };
    38→    case BuildingKind.Services: return { w: 1.5, d: 1.3 };
    39→    case BuildingKind.Cooperative: return { w: 1.4, d: 1.2 };
    40→    case BuildingKind.Market: return { w: 1.5, d: 1.2 };
    41→    case BuildingKind.Castle: return { w: 1.7, d: 1.5 };
    42→    case BuildingKind.Manor: return { w: 1.6, d: 1.4 };
    43→    case BuildingKind.Temple: return { w: 1.5, d: 1.3 };
    44→    case BuildingKind.Academy: return { w: 1.3, d: 1.1 };
    45→    case BuildingKind.Bank: return { w: 1.3, d: 1.1 };
    46→    case BuildingKind.Theater: return { w: 1.4, d: 1.3 };
    47→    case BuildingKind.Hospital: return { w: 1.2, d: 1.1 };
    48→    case BuildingKind.Wall: return { w: 1.6, d: 0.7 };
    49→    default: return { w: 1, d: 1 };
    50→  }
    51→}

    53→/** 等距立方体：中心 (cx,cy)（地面），格尺寸 (gw, gd)，高 hp 像素 */
    54→function box(ctx: CanvasRenderingContext2D, cx: number, cy: number, gw: number, gd: number, hp: number, top: string, left: string, right: string) {
    55→  const hw = (gw * TILE_W) / 2, hd = (gd * TILE_H) / 2;
    56→  // 左面
    57→  ctx.fillStyle = left;
    58→  ctx.beginPath();
    59→  ctx.moveTo(cx - hw, cy);
    60→  ctx.lineTo(cx, cy + hd);
    61→  ctx.lineTo(cx, cy + hd - hp);
    62→  ctx.lineTo(cx - hw, cy - hp);
    63→  ctx.closePath();
    64→  ctx.fill();
    65→  // 右面
    66→  ctx.fillStyle = right;
    67→  ctx.beginPath();
    68→  ctx.moveTo(cx + hw, cy);
    69→  ctx.lineTo(cx, cy + hd);
    70→  ctx.lineTo(cx, cy + hd - hp);
    71→  ctx.lineTo(cx + hw, cy - hp);
    72→  ctx.closePath();
    73→  ctx.fill();
    74→  // 顶面
    75→  ctx.fillStyle = top;
    76→  ctx.beginPath();
    77→  ctx.moveTo(cx, cy - hp - hd);
    78→  ctx.lineTo(cx + hw, cy - hp);
    79→  ctx.lineTo(cx, cy - hp + hd);
    80→  ctx.lineTo(cx - hw, cy - hp);
    81→  ctx.closePath();
    82→  ctx.fill();
    83→  return { topY: cy - hp - hd, hw, hd };
    84→}

    86→/** 双坡屋顶 */
    87→function gableRoof(ctx: CanvasRenderingContext2D, cx: number, cy: number, gw: number, gd: number, hp: number, hrh: number, pal: BuildingPalette, flip = false) {
    88→  const hw = (gw * TILE_W) / 2, hd = (gd * TILE_H) / 2;
    89→  const peak = hp + hrh;
    90→  if (!flip) {
    91→    // 屋脊沿 SW-NE（左右两面）
    92→    ctx.fillStyle = pal.roof;
    93→    ctx.beginPath();
    94→    ctx.moveTo(cx, cy + hd - peak);
    95→    ctx.lineTo(cx + hw, cy - peak);
    96→    ctx.lineTo(cx + hw, cy - hp);
    97→    ctx.lineTo(cx, cy + hd - hp);
    98→    ctx.closePath();
    99→    ctx.fill();
   100→    ctx.fillStyle = pal.roofDark;
   101→    ctx.beginPath();
   102→    ctx.moveTo(cx, cy + hd - peak);
   103→    ctx.lineTo(cx - hw, cy - peak);
   104→    ctx.lineTo(cx - hw, cy - hp);
   105→    ctx.lineTo(cx, cy + hd - hp);
   106→    ctx.closePath();
   107→    ctx.fill();
   108→  } else {
   109→    ctx.fillStyle = pal.roof;
   110→    ctx.beginPath();
   111→    ctx.moveTo(cx, cy - hd - peak + hd);
   112→    ctx.lineTo(cx + hw, cy - peak);
   113→    ctx.lineTo(cx, cy + hd - peak + hd);
   114→    ctx.closePath();
   115→    ctx.fill();
   116→  }
   117→}

   119→function shadow(ctx: CanvasRenderingContext2D, cx: number, cy: number, gw: number, gd: number) {
   120→  const hw = (gw * TILE_W) / 2 + 3, hd = (gd * TILE_H) / 2 + 2;
   121→  ctx.fillStyle = 'rgba(40,52,44,0.22)';
   122→  ctx.beginPath();
   123→  ctx.ellipse(cx + 2, cy + 2, hw * 0.8, hd * 0.8, 0, 0, Math.PI * 2);
   124→  ctx.fill();
   125→}

   127→/** 烟囱烟雾 */
   128→function smoke(ctx: CanvasRenderingContext2D, cx: number, cy: number, topY: number, time: number, seed: number, intensity = 1) {
   129→  const n = 3;
   130→  for (let i = 0; i < n; i++) {
   131→    const ph = (time * 0.0004 + i / n + seed) % 1;
   132→    const a = (1 - ph) * 0.28 * intensity;
   133→    const r = 2.5 + ph * 7;
   134→    ctx.fillStyle = `rgba(220,222,226,${a.toFixed(3)})`;
   135→    ctx.beginPath();
   136→    ctx.arc(cx + Math.sin(ph * 6 + seed * 9) * 4, topY - ph * 26, r, 0, Math.PI * 2);
   137→    ctx.fill();
   138→  }
   139→}

   141→/**
   142→ * 绘制建筑。sx, sy 为建筑所在格的地面世界像素中心。
   143→ * time 毫秒用于动画（炊烟等）。
   144→ */
   145→export function drawBuilding(
   146→  ctx: CanvasRenderingContext2D,
   147→  w: WorldState,
   148→  b: Building,
   149→  sx: number, sy: number,
   150→  time: number,
   151→  lod: number,
   152→) {
   153→  const era = eraOf(w, b);
   154→  const pal = PALETTES[era];
   155→  const fp = footprint(b.kind);
   156→  const seed = vhash(b.x, b.y, 5);
   157→
   158→  // 建造中：脚手架 + 半成品
   159→  if (b.progress < 1) {
   160→    shadow(ctx, sx, sy, fp.w, fp.d);
   161→    ctx.globalAlpha = 0.35 + b.progress * 0.6;
   162→    drawCompleted(ctx, w, b, sx, sy, time, era, pal, seed, lod, true);
   163→    ctx.globalAlpha = 1;
   164→    // 脚手架线条
   165→    ctx.strokeStyle = 'rgba(120,90,56,0.8)';
   166→    ctx.lineWidth = 1.5;
   167→    const hw = (fp.w * TILE_W) / 2, hd = (fp.d * TILE_H) / 2;
   168→    ctx.beginPath();
   169→    ctx.moveTo(sx - hw, sy); ctx.lineTo(sx, sy + hd); ctx.lineTo(sx, sy + hd - 22); ctx.lineTo(sx - hw, sy - 22); ctx.closePath();
   170→    ctx.moveTo(sx + hw, sy); ctx.lineTo(sx, sy + hd); ctx.lineTo(sx, sy + hd - 22); ctx.lineTo(sx + hw, sy - 22); ctx.closePath();
   171→    ctx.stroke();
   172→    return;
   173→  }
   174→  shadow(ctx, sx, sy, fp.w, fp.d);
   175→  drawCompleted(ctx, w, b, sx, sy, time, era, pal, seed, lod, false);
   176→}

   178→function drawCompleted(
   179→  ctx: CanvasRenderingContext2D,
   180→  w: WorldState,
   181→  b: Building,
   182→  sx: number, sy: number,
   183→  time: number,
   184→  era: number,
   185→  pal: BuildingPalette,
   186→  seed: number,
   187→  lod: number,
   188→  ghost: boolean,
   189→) {
   190→  switch (b.kind) {
   191→    case BuildingKind.Camp: {
   192→      // 帐篷 + 篝火
   193→      const h = 16;
   194→      ctx.fillStyle = 'rgba(40,52,44,0.2)';
   195→      ctx.beginPath(); ctx.ellipse(sx, sy + 2, 20, 9, 0, 0, Math.PI * 2); ctx.fill();
   196→      ctx.fillStyle = pal.cloth;
   197→      ctx.beginPath();
   198→      ctx.moveTo(sx - 2, sy + 8);
   199→      ctx.lineTo(sx - 16, sy - h);
   200→      ctx.lineTo(sx + 2, sy - h - 2);
   201→      ctx.closePath();
   202→      ctx.fill();
   203→      ctx.fillStyle = pal.roof;
   204→      ctx.beginPath();
   205→      ctx.moveTo(sx + 2, sy + 8);
   206→      ctx.lineTo(sx + 14, sy - h + 2);
   207→      ctx.lineTo(sx + 2, sy - h - 2);
   208→      ctx.closePath();
   209→      ctx.fill();
   210→      ctx.fillStyle = 'rgba(60,42,26,0.85)';
   211→      ctx.beginPath();
   212→      ctx.moveTo(sx - 2, sy + 8); ctx.lineTo(sx - 9, sy - h + 8); ctx.lineTo(sx + 2, sy - h - 2);
   213→      ctx.closePath(); ctx.fill();
   214→      if (!ghost && lod > 0.5) {
   215→        // 篝火
   216→        const fl = 3 + Math.sin(time * 0.008 + seed * 9) * 1.4;
   217→        ctx.fillStyle = 'rgb(224,120,50)';
   218→        ctx.beginPath();
   219→        ctx.ellipse(sx + 14, sy + 6, 3, fl, 0, 0, Math.PI * 2); ctx.fill();
   220→        ctx.fillStyle = 'rgba(255,206,90,0.75)';
   221→        ctx.beginPath();
   222→        ctx.ellipse(sx + 14, sy + 5, 1.6, fl * 0.55, 0, 0, Math.PI * 2); ctx.fill();
   223→      }
   224→      break;
   225→    }
   226→    case BuildingKind.Shelter: {
   227→      // 窝棚：树枝斜搭 + 茅草（v2.0 更精细：骨架线条 + 茅草层次）
   228→      shadow(ctx, sx, sy, 0.7, 0.7);
   229→      const h = 13;
   230→      ctx.strokeStyle = pal.woodDark;
   231→      ctx.lineWidth = 1.6;
   232→      ctx.beginPath();
   233→      ctx.moveTo(sx - 10, sy + 5); ctx.lineTo(sx - 2, sy - h);
   234→      ctx.moveTo(sx + 10, sy + 5); ctx.lineTo(sx + 2, sy - h);
   235→      ctx.moveTo(sx - 7, sy + 1); ctx.lineTo(sx + 1, sy - h + 3);
   236→      ctx.stroke();
   237→      // 茅草面（两片，带茅草簇）
   238→      ctx.fillStyle = pal.roof;
   239→      ctx.beginPath();
   240→      ctx.moveTo(sx - 2, sy + 7); ctx.lineTo(sx - 13, sy - h + 2); ctx.lineTo(sx + 1, sy - h - 1);
   241→      ctx.closePath(); ctx.fill();
   242→      ctx.fillStyle = pal.roofDark;
   243→      ctx.beginPath();
   244→      ctx.moveTo(sx + 2, sy + 7); ctx.lineTo(sx + 12, sy - h + 3); ctx.lineTo(sx + 2, sy - h - 1);
   245→      ctx.closePath(); ctx.fill();
   246→      // 茅草纹理
   247→      ctx.strokeStyle = 'rgba(120,96,52,0.5)';
   248→      ctx.lineWidth = 0.7;
   249→      ctx.beginPath();
   250→      for (let i = 0; i < 3; i++) {
   251→        ctx.moveTo(sx - 9 + i * 3, sy - h * 0.35 + i);
   252→        ctx.lineTo(sx - 6 + i * 3, sy - h * 0.62 + i);
   253→      }
   254→      ctx.stroke();
   255→      break;
   256→    }
   257→    case BuildingKind.House: {
   258→      // 农舍 v2.0：木板墙纹理 + 双窗 + 门廊
   259→      box(ctx, sx, sy, 0.85, 0.85, 14, pal.wood, pal.woodDark, shade(pal.wood, 0.85));
   260→      // 木板横纹
   261→      ctx.strokeStyle = 'rgba(90,64,42,0.35)';
   262→      ctx.lineWidth = 0.8;
   263→      ctx.beginPath();
   264→      for (let i = 1; i <= 3; i++) {
   265→        const yy = sy - 2 - i * 3;
   266→        ctx.moveTo(sx - 13, yy - 13.5); ctx.lineTo(sx, yy - 19.5);
   267→        ctx.moveTo(sx + 13, yy - 13.5); ctx.lineTo(sx, yy - 19.5);
   268→      }
   269→      ctx.stroke();
   270→      gableRoof(ctx, sx, sy, 0.95, 0.95, 14, 8, pal);
   271→      if (lod > 0.7) {
   272→        ctx.fillStyle = 'rgba(255,214,140,0.85)';
   273→        ctx.fillRect(sx + 4, sy - 8, 4, 4);
   274→        ctx.fillRect(sx - 9, sy - 11, 3.5, 3.5);
   275→        ctx.fillStyle = pal.woodDark;
   276→        ctx.fillRect(sx - 5, sy - 5, 5, 6);
   277→        ctx.fillStyle = 'rgba(40,30,20,0.5)';
   278→        ctx.fillRect(sx - 5, sy - 5, 5, 1.2); // 门楣阴影
   279→      }
   280→      if (!ghost && lod > 0.6) smoke(ctx, sx - 6, sy, sy - 26, time, seed, 0.5);
   281→      break;
   282→    }
   283→    case BuildingKind.Townhouse: {
   284→      box(ctx, sx, sy, 0.8, 0.8, 30, pal.stone, pal.stoneDark, shade(pal.stone, 0.8));
   285→      // 砖缝
   286→      ctx.strokeStyle = 'rgba(90,88,84,0.3)';
   287→      ctx.lineWidth = 0.7;
   288→      ctx.beginPath();
   289→      for (let i = 1; i <= 7; i++) {
   290→        const yy = sy - i * 3.4;
   291→        ctx.moveTo(sx - 12.5, yy - 12); ctx.lineTo(sx, yy - 17.5);
   292→        ctx.moveTo(sx + 12.5, yy - 12); ctx.lineTo(sx, yy - 17.5);
   293→      }
   294→      ctx.stroke();
   295→      gableRoof(ctx, sx, sy, 0.9, 0.9, 30, 9, pal);
   296→      if (lod > 0.65) {
   297→        for (let i = 0; i < 3; i++) {
   298→          ctx.fillStyle = i % 2 ? 'rgba(255,214,140,0.85)' : 'rgba(150,170,180,0.8)';
   299→          ctx.fillRect(sx + 4, sy - 6 - i * 9, 4, 4);
   300→          ctx.fillRect(sx - 8, sy - 6 - i * 9, 4, 4);
   301→        }
   302→      }
   303→      break;
   304→    }
   305→    case BuildingKind.Farm: {
   306→      // 田垄与作物：随年内周期生长
   307→      const growth = cropGrowth(w, b);
   308→      drawField(ctx, sx, sy, growth, seed);
   309→      break;
   310→    }
   311→    case BuildingKind.Granary: {
   312→      // 架高粮仓
   313→      ctx.strokeStyle = pal.woodDark;
   314→      ctx.lineWidth = 2.5;
   315→      ctx.beginPath();
   316→      ctx.moveTo(sx - 12, sy + 4); ctx.lineTo(sx - 12, sy - 6);
   317→      ctx.moveTo(sx + 12, sy + 4); ctx.lineTo(sx + 12, sy - 6);
   318→      ctx.stroke();
   319→      box(ctx, sx, sy - 6, 0.75, 0.75, 10, pal.wood, pal.woodDark, shade(pal.wood, 0.85));
   320→      gableRoof(ctx, sx, sy - 6, 0.85, 0.85, 10, 7, pal);
   321→      if (lod > 0.65) {
   322→        ctx.fillStyle = 'rgb(224,196,120)';
   323→        ctx.beginPath();
   324→        ctx.ellipse(sx + 14, sy + 2, 3.5, 2.2, 0, 0, Math.PI * 2);
   325→        ctx.fill();
   326→      }
   327→      break;
   328→    }
   329→    case BuildingKind.Well: {
   330→      ctx.fillStyle = pal.stone;
   331→      ctx.beginPath();
   332→      ctx.ellipse(sx, sy, 9, 5, 0, 0, Math.PI * 2);
   333→      ctx.fill();
   334→      ctx.fillStyle = 'rgb(72,110,140)';
   335→      ctx.beginPath();
   336→      ctx.ellipse(sx, sy, 5.5, 3, 0, 0, Math.PI * 2);
   337→      ctx.fill();
   338→      ctx.strokeStyle = pal.woodDark;
   339→      ctx.lineWidth = 2;
   340→      ctx.beginPath();
   341→      ctx.moveTo(sx - 7, sy); ctx.lineTo(sx - 4, sy - 14);
   342→      ctx.moveTo(sx + 7, sy); ctx.lineTo(sx + 4, sy - 14);
   343→      ctx.moveTo(sx - 5, sy - 13); ctx.lineTo(sx + 5, sy - 13);
   344→      ctx.stroke();
   345→      ctx.fillStyle = pal.roof;
   346→      ctx.beginPath();
   347→      ctx.moveTo(sx - 8, sy - 13); ctx.lineTo(sx, sy - 19); ctx.lineTo(sx + 8, sy - 13);
   348→      ctx.closePath(); ctx.fill();
   349→      break;
   350→    }
   351→    case BuildingKind.Workshop: {
   352→      box(ctx, sx, sy, 0.9, 0.9, 16, pal.wood, pal.woodDark, shade(pal.wood, 0.82));
   353→      gableRoof(ctx, sx, sy, 1.0, 1.0, 16, 8, pal);
   354→      box(ctx, sx + 10, sy - 4, 0.22, 0.22, 22, pal.stone, pal.stoneDark, pal.stone);
   355→      if (!ghost) smoke(ctx, sx + 10, sy - 4, sy - 30, time, seed);
   356→      if (lod > 0.7) {
   357→        ctx.fillStyle = pal.accent;
   358→        ctx.fillRect(sx - 6, sy - 3, 4, 4); // 工件
   359→      }
   360→      break;
   361→    }
   362→    case BuildingKind.Market: {
   363→      // 帐篷摊位
   364→      const posts: [number, number][] = [[-16, 0], [0, 8], [16, 0], [0, -8]];
   365→      ctx.strokeStyle = pal.woodDark;
   366→      ctx.lineWidth = 2;
   367→      for (const [px, py] of posts) {
   368→        ctx.beginPath(); ctx.moveTo(sx + px, sy + py); ctx.lineTo(sx + px, sy + py - 12); ctx.stroke();
   369→      }
   370→      // 条纹顶
   371→      for (let i = 0; i < 6; i++) {
   372→        ctx.fillStyle = i % 2 ? pal.roof : pal.cloth;
   373→        const t0 = i / 6, t1 = (i + 1) / 6;
   374→        const ax = -16 + t0 * 32, ay = -8 + t0 * 16;
   375→        const bx = -16 + t1 * 32, by = -8 + t1 * 16;
   376→        const cx2 = -16 + t0 * 32, cy2 = 8 - t0 * 16;
   377→        const dx = -16 + t1 * 32, dy = 8 - t1 * 16;
   378→        ctx.beginPath();
   379→        ctx.moveTo(sx + ax, sy + ay - 12);
   380→        ctx.lineTo(sx + bx, sy + by - 12);
   381→        ctx.lineTo(sx + dx, sy + dy - 12);
   382→        ctx.lineTo(sx + cx2, sy + cy2 - 12);
   383→        ctx.closePath();
   384→        ctx.fill();
   385→      }
   386→      if (lod > 0.6) {
   387→        ctx.fillStyle = pal.accent;
   388→        ctx.beginPath(); ctx.ellipse(sx - 6, sy + 3, 3, 2, 0, 0, Math.PI * 2); ctx.fill();
   389→        ctx.fillStyle = 'rgb(178,96,70)';
   390→        ctx.beginPath(); ctx.ellipse(sx + 7, sy + 1, 3, 2, 0, 0, Math.PI * 2); ctx.fill();
   391→      }
   392→      break;
   393→    }
   394→    case BuildingKind.Admin: {
   395→      // 台基 + 柱廊 + 旗帜
   396→      box(ctx, sx, sy, 1.5, 1.3, 6, pal.stone, pal.stoneDark, shade(pal.stone, 0.8));
   397→      box(ctx, sx, sy - 6, 1.3, 1.1, 20, pal.wood, pal.woodDark, shade(pal.wood, 0.8));
   398→      gableRoof(ctx, sx, sy - 6, 1.45, 1.25, 20, 10, pal);
   399→      if (lod > 0.55) {
   400→        ctx.strokeStyle = pal.cloth;
   401→        ctx.lineWidth = 2;
   402→        ctx.beginPath();
   403→        ctx.moveTo(sx, sy - 34);
   404→        ctx.lineTo(sx, sy - 48);
   405→        ctx.stroke();
   406→        const wave = Math.sin(time * 0.004 + seed * 7) * 2;
   407→        ctx.fillStyle = 'rgb(178,86,70)';
   408→        ctx.beginPath();
   409→        ctx.moveTo(sx, sy - 48);
   410→        ctx.lineTo(sx + 12 + wave, sy - 45);
   411→        ctx.lineTo(sx, sy - 40);
   412→        ctx.closePath(); ctx.fill();
   413→      }
   414→      break;
   415→    }
   416→    case BuildingKind.Wall: {
   417→      const fpw = footprint(BuildingKind.Wall);
   418→      box(ctx, sx, sy, fpw.w, fpw.d, 12, pal.stone, pal.stoneDark, shade(pal.stone, 0.78));
   419→      // 城齿

   756→/** 农田：垄沟 + 随周期生长的作物色 */
   757→function drawField(ctx: CanvasRenderingContext2D, sx: number, sy: number, growth: number, seed: number) {
   758→  const hw = TILE_W / 2 - 4, hh = TILE_H / 2 - 2;
   759→  // 土地
   760→  ctx.fillStyle = 'rgb(146,110,72)';
   761→  ctx.beginPath();
   762→  ctx.moveTo(sx, sy - hh); ctx.lineTo(sx + hw, sy); ctx.lineTo(sx, sy + hh); ctx.lineTo(sx - hw, sy);
   763→  ctx.closePath(); ctx.fill();
   764→  // 垄
   765→  ctx.strokeStyle = 'rgb(120,88,56)';
   766→  ctx.lineWidth = 2;
   767→  ctx.beginPath();
   768→  for (let i = -2; i <= 2; i++) {
   769→    const ox = i * hw / 2.6;
   770→    ctx.moveTo(sx + ox, sy - hh / 2 + Math.abs(i) * 1.6);
   771→    ctx.lineTo(sx + ox * 0.2, sy + hh / 2 - Math.abs(i) * 1.2);
   772→  }
   773→  ctx.stroke();
   774→  // 作物
   775→  if (growth > 0.15) {
   776→    const g = Math.min(1, growth);
   777→    const cropColor = g < 0.5 ? `rgb(${120 + g * 60},${150 + g * 50},${70})` : `rgb(${200 + g * 40},${168 - g * 40},${60})`;
   778→    ctx.fillStyle = cropColor;
   779→    for (let i = -2; i <= 2; i++) {
   780→      for (let j = -1; j <= 1; j++) {
   781→        const ox = i * hw / 3 + j * 4, oy = j * hh / 2.4 + i * 1.4;
   782→        const hgt = 2 + g * 7;
   783→        ctx.beginPath();
   784→        ctx.ellipse(sx + ox, sy + oy - hgt * 0.4, 1.6, hgt * 0.55, 0, 0, Math.PI * 2);
   785→        ctx.fill();
   786→      }
   787→    }
   788→  }
   789→}

   791→function cropGrowth(w: WorldState, b: Building): number {
   792→  // 以建造年 + 当前年的小数相位模拟周期生长（纯视觉）
   793→  const phase = (w.year % 3) + ((b.id % 7) / 7);
   794→  return 0.15 + 0.85 * Math.abs(Math.sin(phase * Math.PI / 3));
   795→}

   797→function eraOf(w: WorldState, b: Building): number {
   798→  const t = w.civ.tech.levels;
   799→  const lvl = (d: string) => t[d] ?? 1;
   800→  switch (b.kind) {
   801→    case BuildingKind.Camp: return 0;
   802→    case BuildingKind.Shelter: return 0;
   803→    case BuildingKind.Townhouse: case BuildingKind.Factory: return 3;
   804→    case BuildingKind.Bank: case BuildingKind.Theater: return 3;
   805→    case BuildingKind.Cooperative: case BuildingKind.Services: case BuildingKind.Hospital: return 4;
   806→    case BuildingKind.Temple: case BuildingKind.Manor: case BuildingKind.Castle: return 2;
   807→    case BuildingKind.Academy: return lvl('education') >= 4 ? 3 : 2;
   808→    case BuildingKind.Market: case BuildingKind.Admin: case BuildingKind.Wall: case BuildingKind.Workshop: return 2;
   809→    default: return lvl('agriculture') >= 2 ? 1 : 0;
   810→  }
   811→}
