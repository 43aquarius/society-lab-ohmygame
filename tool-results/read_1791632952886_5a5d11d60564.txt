    37→function openModal(html: string, wide = false) {
    38→  const root = document.getElementById('modal-root')!;
    39→  root.innerHTML = `
    40→    <div class="modal-backdrop"></div>
    41→    <div class="modal ${wide ? 'wide' : ''}">${html}</div>`;
    42→  modalOpen = true;
    43→  root.querySelector('.modal-backdrop')!.addEventListener('click', closeModal);
    44→  const closeBtn = root.querySelector('.modal-close');
    45→  if (closeBtn) closeBtn.addEventListener('click', closeModal);
    46→  return root.querySelector('.modal') as HTMLElement;
    47→}

    51→export function openEventCard(ev: EventRecord, w: WorldState, actions: PanelActions) {
    52→  const meta = EVENT_ICON[ev.category] ?? EVENT_ICON.harvest;
    53→  const settlement = ev.settlementId !== undefined ? w.settlements.find(s => s.id === ev.settlementId) : undefined;
    54→  let actionsHtml = '';
    55→  if (ev.category === 'famine' && settlement) {
    56→    actionsHtml = `<button class="btn gold" data-act="bless">${icon('blessing', 15)} 赐下丰产祝福（12⚡）</button>`;
    57→  }
    58→  if (ev.category === 'plague' && settlement && w.activeEvents.some(a => a.id === 'plague-outbreak' && a.data.settlementId === settlement.id)) {
    59→    actionsHtml = `<button class="btn gold" data-act="purify">${icon('sparkle', 15)} 净化疫病（12⚡）</button>`;
    60→  }
    61→  if (ev.category === 'ideas' || ev.category === 'institution') {
    62→    actionsHtml = `<button class="btn" data-act="ideas">${icon('ideas', 15)} 传播组织与平等思想（14⚡）</button>`;
    63→  }
    64→  const modal = openModal(`
    65→    <div class="modal-head">
    66→      <span class="mh-ic" style="background:${meta.color}">${icon(meta.icon, 18)}</span>
    67→      <h2>${ev.title}</h2>
    68→      <button class="modal-close">${icon('close', 16)}</button>
    69→    </div>
    70→    <div class="modal-body">
    71→      <div class="event-card-text">${ev.text}</div>
    72→      <div class="ce-section">
    73→        <h5>${icon('chart', 13)} 事件起因（真实模拟变量）</h5>
    74→        <ul>${ev.causes.map(c => `<li>${c}</li>`).join('')}</ul>
    75→      </div>
    76→      <div class="ce-section effects">
    77→        <h5>${icon('sparkle', 13)} 实际结果</h5>
    78→        <ul>${ev.effects.map(e => `<li>${e}</li>`).join('')}</ul>
    79→      </div>
    80→      <div class="ce-section">
    81→        <h5>${icon('clock', 13)} 时间地点</h5>
    82→        <ul><li>第 ${ev.year} 年${settlement ? ` · ${settlement.name}` : ' · 全域'}</li>
    83→        <li>${ev.intervened ? '神（玩家）介入了此事' : '神选择旁观'}</li></ul>
    84→      </div>
    85→      ${actionsHtml ? `<div class="ce-actions">${actionsHtml}<button class="btn" data-act="close">静观其变</button></div>` : ''}
    86→    </div>`);
    87→  modal.querySelector('[data-act="close"]')?.addEventListener('click', closeModal);
    88→  modal.querySelector('[data-act="bless"]')?.addEventListener('click', () => {
    89→    if (settlement) actions.castBlessingAt(settlement);
    90→    closeModal();
    91→  });
    92→  modal.querySelector('[data-act="purify"]')?.addEventListener('click', () => {
    93→    if (settlement) actions.purify(settlement.id);
    94→    closeModal();
    95→  });
    96→  modal.querySelector('[data-act="ideas"]')?.addEventListener('click', () => {
    97→    const def = GOD_POWERS.find(p => p.id === 'ideas')!;
    98→    if (!canAfford(w, def.cost)) { actions.toast('神力不足', 'bad', 'alert'); return; }
    99→    castCivPower(w, 'ideas');
   100→    w.chronicle.push({
   101→      id: `ev-${w.nextEventId++}`, year: w.year, title: '神力：思想传播',
   102→      category: 'ideas', text: '平等与组织的观念随神启传遍大地。', causes: ['神（玩家）传播思想'],
   103→      effects: ['政治参与上升', '平民组织能力上升'], intervened: true, severity: 0.2,
   104→    });
   105→    actions.toast('思想已传播', 'good', 'ideas');
   106→    actions.playSfx('power');
   107→    actions.refreshHud();
   108→    closeModal();
   109→  });
   110→}

   114→export function openChronicle(w: WorldState, actions: PanelActions) {
   115→  let filter = 'all';
   116→  const render = () => {
   117→    const entries = [...w.chronicle].reverse().filter(e => {
   118→      if (filter === 'all') return true;
   119→      if (filter === 'major') return e.severity >= 0.5;
   120→      if (filter === 'god') return e.intervened;
   121→      if (filter === 'institution') return e.category === 'institution' || e.category === 'politics';
   122→      return true;
   123→    });
   124→    // 时代分组：按形态变迁分段
   125→    const eras: { label: string; items: ChronicleEntry[] }[] = [];
   126→    let curForm = w.civ.form;
   127→    let bucket: ChronicleEntry[] = [];
   128→    const formNames: Record<string, string> = {
   129→      primitive: '原始社会', slave: '奴隶制社会', feudal: '封建社会',
   130→      capitalist: '资本主义社会', socialist: '社会主义社会', collapsed: '文明废墟',
   131→    };
   132→    const ordered = [...entries].reverse();
   133→    for (const e of ordered) {
   134→      bucket.push(e);
   135→    }
   136→    // 简化：直接按年份倒序 + 时代头（用 formSince 推断）
   137→    eras.push({ label: '全部记录', items: bucket.reverse() });
   138→    const modal = openModal(`
   139→      <div class="modal-head">
   140→        <span class="mh-ic" style="background:#2f7e76">${icon('book', 18)}</span>
   141→        <h2>历史编年史 · ${w.chronicle.length} 条</h2>
   142→        <button class="modal-close">${icon('close', 16)}</button>
   143→      </div>
   144→      <div class="modal-body">
   145→        <div class="chron-toolbar">
   146→          <button data-f="all" class="${filter === 'all' ? 'active' : ''}">全部</button>
   147→          <button data-f="major" class="${filter === 'major' ? 'active' : ''}">重大</button>
   148→          <button data-f="institution" class="${filter === 'institution' ? 'active' : ''}">制度变迁</button>
   149→          <button data-f="god" class="${filter === 'god' ? 'active' : ''}">神迹</button>
   150→        </div>
   151→        ${entries.length === 0 ? '<div class="chron-empty">历史尚未开始书写。<br>推进时间，或施加你的第一个神迹。</div>' : ''}
   152→        ${entries.map(e => {
   153→          const meta = EVENT_ICON[e.category] ?? EVENT_ICON.harvest;
   154→          return `<div class="chron-item" data-eid="${e.id}">
   155→            <div class="c-year">${e.year}</div>
   156→            <div class="c-ic" style="background:${meta.color}">${icon(meta.icon, 15)}</div>
   157→            <div class="c-main">
   158→              <div class="c-title">${e.title}${e.intervened ? '<span class="intervened">神迹</span>' : ''}</div>
   159→              <div class="c-text">${e.text.length > 66 ? e.text.slice(0, 66) + '…' : e.text}</div>
   160→            </div>
   161→          </div>`;
   162→        }).join('')}
   163→      </div>`, true);
   164→    modal.querySelectorAll('.chron-toolbar button').forEach(b => {
   165→      b.addEventListener('click', () => {
   166→        filter = (b as HTMLElement).dataset.f!;
   167→        render();
   168→      });
   169→    });
   170→    modal.querySelectorAll('.chron-item').forEach(el => {
   171→      el.addEventListener('click', () => {
   172→        const e = w.chronicle.find(x => x.id === (el as HTMLElement).dataset.eid);
   173→        if (e) openEventCard(e, w, actions);
   174→      });
   175→    });
   176→  };
   177→  render();
   178→}

   182→function drawLineChart(canvas: HTMLCanvasElement, data: { year: number; v: number }[], color: string, min?: number, max?: number, ref1?: number) {
   183→  const ctx = canvas.getContext('2d')!;
   184→  const W = canvas.width = canvas.clientWidth * 2 || 600;
   185→  const H = canvas.height = 220;
   186→  ctx.scale(1, 1);
   187→  ctx.clearRect(0, 0, W, H);
   188→  if (data.length < 2) {
   189→    ctx.fillStyle = 'rgba(107,97,84,0.5)';
   190→    ctx.font = '22px sans-serif';
   191→    ctx.textAlign = 'center';
   192→    ctx.fillText('数据不足（推进时间以采集）', W / 2, H / 2);
   193→    return;
   194→  }
   195→  const vs = data.map(d => d.v);
   196→  const lo = min ?? Math.min(...vs) * 0.95;
   197→  const hi = max ?? Math.max(...vs) * 1.05;
   198→  const x0 = 40, x1 = W - 14, y0 = 18, y1 = H - 26;
   199→  const px = (i: number) => x0 + (i / (data.length - 1)) * (x1 - x0);
   200→  const py = (v: number) => y1 - ((v - lo) / Math.max(0.0001, hi - lo)) * (y1 - y0);
   201→  // 网格
   202→  ctx.strokeStyle = 'rgba(59,49,40,0.09)';
   203→  ctx.lineWidth = 1;
   204→  for (let i = 0; i <= 3; i++) {
   205→    const y = y0 + (i / 3) * (y1 - y0);
   206→    ctx.beginPath(); ctx.moveTo(x0, y); ctx.lineTo(x1, y); ctx.stroke();
   207→  }
   208→  if (ref1 !== undefined) {
   209→    ctx.strokeStyle = 'rgba(180,85,45,0.5)';
   210→    ctx.setLineDash([5, 5]);
   211→    ctx.beginPath(); ctx.moveTo(x0, py(ref1)); ctx.lineTo(x1, py(ref1)); ctx.stroke();
   212→    ctx.setLineDash([]);
   213→  }
   214→  // 面积
   215→  ctx.beginPath();
   216→  ctx.moveTo(px(0), py(vs[0]));
   217→  vs.forEach((v, i) => ctx.lineTo(px(i), py(v)));
   218→  ctx.lineTo(px(vs.length - 1), y1);
   219→  ctx.lineTo(px(0), y1);
   220→  ctx.closePath();
   221→  ctx.fillStyle = color.replace(')', ',0.13)').replace('rgb', 'rgba');
   222→  ctx.fill();
   223→  // 线
   224→  ctx.beginPath();
   225→  vs.forEach((v, i) => i === 0 ? ctx.moveTo(px(0), py(v)) : ctx.lineTo(px(i), py(v)));
   226→  ctx.strokeStyle = color;
   227→  ctx.lineWidth = 3;
   228→  ctx.stroke();
   229→  // 首尾年份
   230→  ctx.fillStyle = 'rgba(107,97,84,0.85)';
   231→  ctx.font = '20px sans-serif';
   232→  ctx.textAlign = 'left';
   233→  ctx.fillText(`${data[0].year}`, x0, H - 6);
   234→  ctx.textAlign = 'right';
   235→  ctx.fillText(`${data[data.length - 1].year}`, x1, H - 6);
   236→  ctx.fillText(`${vs[vs.length - 1]}`, x1, y0 + 6);
   237→}

   239→function drawStackChart(canvas: HTMLCanvasElement, data: { year: number; elite: number }[]) {
   240→  const ctx = canvas.getContext('2d')!;
   241→  const W = canvas.width = canvas.clientWidth * 2 || 600;
   242→  const H = canvas.height = 220;
   243→  ctx.clearRect(0, 0, W, H);
   244→  if (data.length < 2) {
   245→    ctx.fillStyle = 'rgba(107,97,84,0.5)';
   246→    ctx.font = '22px sans-serif'; ctx.textAlign = 'center';
   247→    ctx.fillText('数据不足', W / 2, H / 2);
   248→    return;
   249→  }
   250→  const x0 = 40, x1 = W - 14, y0 = 18, y1 = H - 26;
   251→  const px = (i: number) => x0 + (i / (data.length - 1)) * (x1 - x0);
   252→  const barW = Math.max(4, (x1 - x0) / data.length * 0.8);
   253→  for (let i = 0; i < data.length; i++) {
   254→    const e = data[i].elite;
   255→    const mid = Math.max(0.05, 1 - e - 0.35);
   256→    const common = Math.max(0, 1 - e - mid);
   257→    const ye = y1 - e * (y1 - y0);
   258→    const ym = ye - mid * (y1 - y0);
   259→    ctx.fillStyle = '#c9a227';
   260→    ctx.fillRect(px(i) - barW / 2, ye, barW, y1 - ye);
   261→    ctx.fillStyle = '#7a8a9a';
   262→    ctx.fillRect(px(i) - barW / 2, ym, barW, ye - ym);
   263→    ctx.fillStyle = '#5d8a6a';
   264→    ctx.fillRect(px(i) - barW / 2, ym - common * (y1 - y0), barW, common * (y1 - y0));
   265→  }
   266→  ctx.fillStyle = 'rgba(107,97,84,0.85)';
   267→  ctx.font = '20px sans-serif';
   268→  ctx.textAlign = 'left';
   269→  ctx.fillText('财富分配：金=精英 灰=中间 绿=平民', x0, y0 + 4);
   270→}

   415→        </div>
   416→        ${s ? `<button class="btn small" data-load="${i}">读取</button>` : ''}
   417→      </div>`;
   418→    }).join('');
   419→    const snaps = w.snapshots.map((s, i) => `
   420→      <div class="snapshot-row">
   421→        <span class="s-ic ${s.data ? 'has-data' : ''}">${icon(s.data ? 'pin' : 'clock', 15)}</span>

   423→        ${s.data ? `<button class="btn small" data-branch="${i}">从此分支</button>` : ''}
   424→      </div>`).join('');
   425→    const modal = openModal(`
   426→      <div class="modal-head">
   427→        <span class="mh-ic" style="background:#5a6a8a">${icon('save', 18)}</span>
   428→        <h2>保存 · 读取 · 分支</h2>
   429→        <button class="modal-close">${icon('close', 16)}</button>
   430→      </div>
   431→      <div class="modal-body">
   432→        <div style="display:flex;gap:10px;margin-bottom:14px;flex-wrap:wrap">
   433→          <button class="btn primary" id="snap-pin">${icon('pin', 15)} 在此记录分支节点</button>
   434→          <button class="btn" id="save-export">${icon('load', 15)} 导出为文件</button>
   435→          <label class="btn" style="cursor:pointer">${icon('save', 15)} 导入存档文件
   436→            <input type="file" id="save-import" accept=".json" style="display:none">
   437→          </label>
   438→          <button class="btn" id="save-autosave" ${localStorage.getItem('society-lab-save-v1-auto') ? '' : 'disabled'}>读取自动存档</button>
   439→        </div>
   440→        <div class="slot-list">${slotEls}</div>
   441→        <h5 style="margin:18px 0 8px;font-size:12.5px;color:var(--ink-soft)">历史节点（每 25 年自动记录 · ◆ 可创建平行世界）</h5>
   442→        ${snaps || '<div class="chron-empty">暂无快照</div>'}
   443→      </div>`, true);
   444→    modal.querySelectorAll('.slot-item').forEach(el => {
   445→      el.addEventListener('click', () => {
   446→        const i = Number((el as HTMLElement).dataset.slot);
   447→        if (saveToSlot(w, worldName, i)) {
   448→          actions.toast(`已存入槽位 ${i}`, 'good', 'save');
   449→          actions.playSfx('click');
   450→          render();
   451→        } else actions.toast('存档失败（空间不足？）', 'bad', 'alert');
   452→      });
   453→    });
   454→    modal.querySelectorAll('[data-load]').forEach(b => {
   455→      b.addEventListener('click', e => {
   456→        e.stopPropagation();
   457→        const res = loadFromSlot(Number((b as HTMLElement).dataset.load));
   458→        if (res && 'world' in res) {
   459→          actions.loadWorld(res.world, res.worldName);
   460→          closeModal();
   461→          actions.toast('世界已读取', 'good', 'load');
   462→        } else {
   463→          actions.toast((res as any).error ?? '读取失败', 'bad', 'alert');
   464→        }
   465→      });
   466→    });
   467→    modal.querySelectorAll('[data-branch]').forEach(b => {
   468→      b.addEventListener('click', () => {
   469→        const res = branchFromSnapshot(w, Number((b as HTMLElement).dataset.branch));
   470→        if (res && 'world' in res) {
   471→          actions.loadWorld(res.world, res.worldName);
   472→          closeModal();
   473→          actions.toast('已创建平行历史分支', 'good', 'branch');
   474→        } else {
   475→          actions.toast((res as any).error ?? '无法分支', 'bad', 'alert');
   476→        }
   477→      });
   478→    });
   479→    modal.querySelector('#snap-pin')!.addEventListener('click', () => {
   480→      pinSnapshot(w, `${w.year} 年 · 手动节点`);
   481→      actions.toast('分支节点已记录', 'good', 'pin');
   482→      render();
   483→    });
   484→    modal.querySelector('#save-export')!.addEventListener('click', () => {
   485→      exportToFile(w, worldName);
   486→      actions.toast('已导出存档文件', 'good', 'save');
   487→    });
   488→    modal.querySelector('#save-import')!.addEventListener('change', async e => {
   489→      const file = (e.target as HTMLInputElement).files?.[0];
   490→      if (!file) return;
   491→      const res = await importFromFile(file);
   492→      if ('world' in res) {
   493→        actions.loadWorld(res.world, res.worldName);
   494→        closeModal();
   495→        actions.toast('存档已导入', 'good', 'load');
   496→      } else actions.toast(res.error, 'bad', 'alert');
   497→    });
   498→    modal.querySelector('#save-autosave')!.addEventListener('click', () => {
   499→      const raw = localStorage.getItem('society-lab-save-v1-auto');
   500→      if (!raw) return;
   501→      const d = deserialize(raw);
   502→      if ('world' in d) {
   503→        actions.loadWorld(d.world, d.worldName);
   504→        closeModal();
   505→        actions.toast('自动存档已读取', 'good', 'load');
   506→      }
   507→    });
   508→  };
   509→  render();
   510→}
