    29→// 屏幕像素 → 浮点网格坐标（供居民检视）
    30→function screenToGridFloat(vw: VisualWorld, px: number, py: number): { x: number; y: number } | null {
    31→  const rect = vw.canvas.getBoundingClientRect();
    32→  const cx = rect.width / 2, cy = rect.height / 2;
    33→  const wx = (px - rect.left - cx) / vw.camera.zoom + vw.camera.x;
    34→  const wy = (py - rect.top - cy) / vw.camera.zoom + vw.camera.y;
    35→  return screenToGrid(wx, wy + 16);
    36→}

   304→  private hudCallbacks(): HudCallbacks {
   305→    return {
   306→      onSpeed: s => {
   307→        this.speed = s;
   308→        this.audio.ensure();
   309→        setHudSpeed(this.els.topbar, s);
   310→        if (this.world) this.showHint(s === 0 ? '时间暂停' : s === 1 ? '正常速度' : s === 2 ? '快速推进' : '极速推进', 1600);
   311→      },
   312→      onStep: () => {
   313→        if (this.world && this.speed === 0) {
   314→          const report = tickYear(this.world);
   315→          this.afterYear(report);
   316→        }
   317→      },
   318→      onPower: id => {
   319→        if (!this.world) return;
   320→        if (id === 'enlighten' || id === 'ideas' || id === 'heal') {
   321→          const res = castCivPower(this.world, id);
   322→          if (res.ok) {
   323→            const entry = powerChronicleEntry(this.world, id, res);
   324→            this.world.chronicle.push(entry);
   325→            this.audio.power();
   326→            this.flash('#7fc7ad', 0.35);
   327→            const s = this.world.settlements.find(x => !x.dead);
   328→            if (s && id === 'heal') this.vw?.fx.spawn('heal', s.x, s.y, 6);
   329→            toast(res.message, 'good', id === 'enlighten' ? 'enlighten' : id === 'heal' ? 'heal' : 'ideas');
   330→            this.refreshHud();
   331→          } else toast(res.message, 'bad', 'alert');
   332→          return;
   333→        }
   334→        this.activePower = this.activePower === id ? null : id;
   335→        setHudPower(this.els.toolbar, this.activePower);
   336→        const def = GOD_POWERS.find(p => p.id === id);
   337→        this.showHint(this.activePower
   338→          ? `神力就绪：${def?.name ?? ''} · 点击地图施放（右键/Esc 取消）${def?.brush ? ' · 可调笔刷' : ''}`
   339→          : '检视模式：点击聚落查看详情', 2600);
   340→      },
   341→      onBrush: r => {
   342→        this.brush = r;
   343→        setHudBrush(this.els.toolbar, r);
   344→        if (this.vw) this.vw.brushRadius = r;
   345→      },
   346→      onChronicle: () => this.world && openChronicle(this.world, this.panelActions()),
   347→      onObserver: () => this.world && openObserver(this.world),
   348→      onSave: () => this.world && openSaveMenu(this.world, this.worldName, this.panelActions()),
   349→      onMenu: () => {
   350→        if (this.world) { autosave(this.world, this.worldName); toast('已自动保存，返回主菜单', 'good', 'save'); }
   351→        this.showMenu();
   352→      },
   353→      onHelp: () => openHelp(),
   354→      onFormBadge: () => this.world && openFormCard(this.world),
   355→      onPopStat: () => this.world && openObserver(this.world),
   356→      onFoodStat: () => this.world && openObserver(this.world),
   357→      onMute: () => {
   358→        this.audio.setMuted(!this.audio.muted);
   359→        setMuteIcon(this.els.topbar, this.audio.muted);
   360→      },
   361→      onHome: () => this.vw && this.world && this.vw.fitMap(this.world.config.size),
   362→      onCollapseToolbar: (collapsed) => {
   363→        localStorage.setItem('society-lab-tb-collapsed', collapsed ? '1' : '0');
   364→      },
   365→    };
   366→  }

   380→  private blessingAt(s: Settlement) {
   381→    if (!this.world) return;
   382→    const res = castBrushPower(this.world, 'blessing', s.x, s.y, 0);
   383→    if (res.ok) {
   384→      const entry = powerChronicleEntry(this.world, 'blessing', res);
   385→      this.world.chronicle.push(entry);
   386→      this.vw?.addPing(s.x, s.y, 'rgba(232,196,98,0.95)');
   387→      this.audio.power();
   388→      this.flash('#e8c462', 0.3);
   389→      toast(`丰产祝福降临 ${s.name}`, 'good', 'blessing');
   390→      this.refreshHud();
   391→    } else toast(res.message, 'bad', 'alert');
   392→  }

   394→  private purifyAt(settlementId: number) {
   395→    if (!this.world) return false;
   396→    const res = castPurify(this.world, settlementId);
   397→    if (res.ok) {
   398→      this.world.chronicle.push({
   399→        id: `ev-${this.world.nextEventId++}`, year: this.world.year, title: '神力：净化疫病',
   400→        category: 'plague', text: '神光扫过病榻，高热退去，咳嗽止息。医者们把这一夜记录了下来。',
   401→        causes: ['神（玩家）净化疫病'], effects: ['疫病终止', '医学知识积累'], intervened: true,
   402→        severity: 0.4, settlementId,
   403→      });
   404→      const s = this.world.settlements.find(x => x.id === settlementId);
   405→      if (s) this.vw?.addPing(s.x, s.y, 'rgba(127,199,173,0.95)');
   406→      this.audio.power();
   407→      this.flash('#7fc7ad', 0.35);
   408→      toast('疫病已被净化', 'good', 'sparkle');
   409→      this.refreshHud();
   410→      return true;
   411→    }
   412→    toast(res.message, 'bad', 'alert');
   413→    return false;
   414→  }

   505→  private handleClick(px: number, py: number, button: number) {
   506→    if (!this.world || !this.vw || button !== 0) return;
   507→    const tile = this.vw.screenToTile(px, py);
   508→    if (!tile) return;
   509→    const size = this.world.config.size;
   510→    if (tile.x < 0 || tile.y < 0 || tile.x >= size || tile.y >= size) {
   511→      // 神力就绪时点到沙盘外给出提示，避免“点了没反应”的困惑（v1.0.1）
   512→      if (this.activePower) this.showHint('点击在沙盘之外 · 请点在地图上施放', 1800);
   513→      return;
   514→    }
   515→
   516→    if (this.activePower) {
   517→      const res = castBrushPower(this.world, this.activePower, tile.x, tile.y, this.brush);
   518→      if (res.ok) {
   519→        // 陨星在 castMeteor 内部已记录编年史，不重复记录
   520→        if (this.activePower !== 'meteor') {
   521→          const entry = powerChronicleEntry(this.world, this.activePower, res);
   522→          this.world.chronicle.push(entry);
   523→        }
   524→        // 地形变化 → 重建缓存与小地图
   525→        const terrainPowers: GodPowerId[] = ['forest', 'fertile', 'meadow', 'water', 'ore', 'uplift', 'meteor', 'wildfire'];
   526→        if (terrainPowers.includes(this.activePower)) {
   527→          this.vw.terrainCache.dirty = true;
   528→          this.vw.minimapDirty = true;
   529→        }
   530→        // 特效
   531→        this.vw.fx.spawnForPower(this.activePower, tile.x, tile.y, this.world);
   532→        this.vw.addPing(tile.x, tile.y, this.activePower === 'drought' || this.activePower === 'meteor' ? 'rgba(220,120,80,0.95)' : 'rgba(127,199,173,0.9)');
   533→        this.audio.power();
   534→        if (this.activePower === 'meteor') this.flash('#ffb060', 0.4);
   535→        else this.flash(this.activePower === 'drought' ? 'rgb(216,160,80)' : 'rgb(127,199,173)', 0.28);
   536→        toast(res.message, 'good', this.activePower);
   537→        this.refreshHud();
   538→      } else {
   539→        toast(res.message, 'bad', 'alert');
   540→      }
   541→      return;
   542→    }
   543→    // 检视模式：先看是否点中了居民（真实的人，可查看其生平）
   544→    const g = screenToGridFloat(this.vw, px, py);
   545→    if (g) {
   546→      const agent = this.vw.agents.pick(g.x, g.y);
   547→      if (agent) {
   548→        const s = this.world.settlements.find(x => x.id === agent.sid);
   549→        const stage = agent.age < CFG.agents.adultAge ? '孩童'
   550→          : agent.age >= CFG.agents.elderAge ? '长者' : '成年';
   551→        const health = agent.sick > 0.3 ? '⚠ 正在患病' : '健康';
   552→        const doing = agent.state === 'work' ? '正在劳作'
   553→          : agent.state === 'sleep' ? '已入睡'
   554→            : agent.state === 'goto' ? '在路上'
   555→              : agent.state === 'social' || agent.state === 'idle' ? '歇息/交谈'
   556→                : agent.state === 'migrate' ? '迁往新居' : '待在' + (s?.name ?? '');
   557→        toast(`${agentName(agent.id)} · ${agent.gender ? '女' : '男'} · ${agent.age} 岁（${stage}） · ${jobName(agent.job, agent.age)} · ${health} · ${doing}`, 'neutral', 'people');
   558→        this.vw.addPing(Math.round(agent.gx), Math.round(agent.gy), 'rgba(242,234,214,0.95)');
   559→        this.audio.click();
   560→        return;
   561→      }
   562→    }
   563→    // 找最近聚落
   564→    let best: Settlement | null = null, bestD = 2.4;
   565→    for (const s of this.world.settlements) {
   566→      const d = Math.hypot(s.x - tile.x, s.y - tile.y);
   567→      if (d < bestD) { bestD = d; best = s; }
   568→    }
   569→    if (best) {
   570→      showSettlementPanel(this.els.sidepanel, this.world, best, this.panelActions());
   571→      this.audio.click();
   572→    } else {
   573→      hideSettlementPanel(this.els.sidepanel);
   574→    }
   575→  }

   641→  frame(time: number) {
   642→    const dt = Math.min(0.06, (time - this.lastTime) / 1000 || 0.016);
   643→    this.lastTime = time;
   644→
   645→    // 菜单背景
   646→    if ((document.getElementById('menu') as HTMLElement).style.display !== 'none' && this.menuVw && this.menuWorld) {
   647→      const cam = this.menuVw.camera;
   648→      cam.targetX = Math.sin(time * 0.00008) * 380;
   649→      cam.targetY = this.menuWorld.config.size * 16 + Math.cos(time * 0.00006) * 160;
   650→      this.menuVw.render(this.menuWorld, dt, time, true);
   651→    }
   652→
   653→    // 游戏世界
   654→    if (this.world && this.vw && (document.getElementById('game') as HTMLElement).classList.contains('hidden') === false) {
   655→      if (this.speed > 0 && !isModalOpen()) {
   656→        this.yearAccum += dt * this.speed;
   657→        const yearSeconds = CFG.yearMs1x / 1000;
   658→        while (this.yearAccum >= yearSeconds) {
   659→          this.yearAccum -= yearSeconds;
   660→          const report = tickYear(this.world);
   661→          this.afterYear(report);
   662→          if (this.speed === 0) break;
   663→        }
   664→      }
   665→      this.vw.render(this.world, dt, time, this.speed === 0);
   666→    }
   667→    this.raf = requestAnimationFrame(t => this.frame(t));
   668→  }

   670→  afterYear(report: ReturnType<typeof tickYear>) {
   671→    if (!this.world) return;
   672→    const w = this.world;
   673→    this.refreshHud();
   674→
   675→    // v3.0：居民智能体随年度同步（生老病死/迁徙/职业/时疫）
   676→    this.vw?.agents.syncYear(w);
   677→
   678→    // 事件提示 + 视觉特效
   679→    for (const ev of report.events) {
   680→      if (ev.category === 'institution') {
   681→        this.playSfx('era');
   682→        toast(ev.title, 'good', 'branch');
   683→        const s = ev.settlementId !== undefined ? w.settlements.find(x => x.id === ev.settlementId) : null;
   684→        if (s) this.vw?.addPing(s.x, s.y, 'rgba(201,162,39,0.95)');
   685→      } else if (ev.severity > 0.55) {
   686→        this.playSfx('chime', ev.severity);
   687→        toast(ev.title, 'bad', 'alert');
   688→        const s = ev.settlementId !== undefined ? w.settlements.find(x => x.id === ev.settlementId) : null;
   689→        if (s) this.vw?.addPing(s.x, s.y, 'rgba(180,85,45,0.9)');
   690→      } else if (ev.category === 'invention' || ev.category === 'achievement') {
   691→        this.playSfx('milestone');
   692→      }
   693→      // 事件 → 沙盘上的真实视觉表现（v2.0）
   694→      this.vw?.fx.spawnForEvent(w, ev);
   695→    }
   696→
   697→    // 事件流
   698→    updateEventFeed(this.els.feed, w, this.panelActions(), this.lastFeedYear);
   699→    this.lastFeedYear = w.year;
   700→
   701→    // 雨声强度
   702→    if (this.vw && this.world) {
   703→      let wetCount = 0;
   704→      const step = 3;
   705→      for (let i = 0; i < this.world.weather.length; i += step * step) {
   706→        if (this.world.weather[i].wet > 0.5) wetCount++;
   707→      }
   708→      this.audio.setRainIntensity(Math.min(1, wetCount / 40));
   709→    }
   710→
   711→    // 自动存档（每 10 年）
   712→    if (w.year % 10 === 0) autosave(w, this.worldName);
   713→
   714→    // 文明湮灭提示
   715→    if (w.civ.vars.population === 0 && w.year > 10) {
   716→      this.speed = 0;
   717→      setHudSpeed(this.els.topbar, 0);
   718→      toast('文明已经湮灭……可读取存档或从分支重启', 'bad', 'alert');
   719→    }
   720→  }
