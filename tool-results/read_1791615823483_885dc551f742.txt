   292→  private hudCallbacks(): HudCallbacks {
   293→    return {
   294→      onSpeed: s => {
   295→        this.speed = s;
   296→        this.audio.ensure();
   297→        setHudSpeed(this.els.topbar, s);
   298→        if (this.world) this.showHint(s === 0 ? '时间暂停' : s === 1 ? '正常速度' : s === 2 ? '快速推进' : '极速推进', 1600);
   299→      },
   300→      onStep: () => {
   301→        if (this.world && this.speed === 0) {
   302→          const report = tickYear(this.world);
   303→          this.afterYear(report);
   304→        }
   305→      },
   306→      onPower: id => {
   307→        if (!this.world) return;
   308→        if (id === 'enlighten' || id === 'ideas' || id === 'heal') {
   309→          const res = castCivPower(this.world, id);
   310→          if (res.ok) {
   311→            const entry = powerChronicleEntry(this.world, id, res);
   312→            this.world.chronicle.push(entry);
   313→            this.audio.power();
   314→            this.flash('#7fc7ad', 0.35);
   315→            const s = this.world.settlements.find(x => !x.dead);
   316→            if (s && id === 'heal') this.vw?.fx.spawn('heal', s.x, s.y, 6);
   317→            toast(res.message, 'good', id === 'enlighten' ? 'enlighten' : id === 'heal' ? 'heal' : 'ideas');
   318→            this.refreshHud();
   319→          } else toast(res.message, 'bad', 'alert');
   320→          return;
   321→        }
   322→        this.activePower = this.activePower === id ? null : id;
   323→        setHudPower(this.els.toolbar, this.activePower);
   324→        const def = GOD_POWERS.find(p => p.id === id);
   325→        this.showHint(this.activePower
   326→          ? `神力就绪：${def?.name ?? ''} · 点击地图施放（右键/Esc 取消）${def?.brush ? ' · 可调笔刷' : ''}`
   327→          : '检视模式：点击聚落查看详情', 2600);
   328→      },
   329→      onBrush: r => {
   330→        this.brush = r;
   331→        setHudBrush(this.els.toolbar, r);
   332→        if (this.vw) this.vw.brushRadius = r;
   333→      },
   334→      onChronicle: () => this.world && openChronicle(this.world, this.panelActions()),
   335→      onObserver: () => this.world && openObserver(this.world),
   336→      onSave: () => this.world && openSaveMenu(this.world, this.worldName, this.panelActions()),
   337→      onMenu: () => {
   338→        if (this.world) { autosave(this.world, this.worldName); toast('已自动保存，返回主菜单', 'good', 'save'); }
   339→        this.showMenu();
   340→      },
   341→      onHelp: () => openHelp(),
   342→      onFormBadge: () => this.world && openFormCard(this.world),
   343→      onPopStat: () => this.world && openObserver(this.world),
   344→      onFoodStat: () => this.world && openObserver(this.world),
   345→      onMute: () => {
   346→        this.audio.setMuted(!this.audio.muted);
   347→        setMuteIcon(this.els.topbar, this.audio.muted);
   348→      },
   349→      onHome: () => this.vw && this.world && this.vw.fitMap(this.world.config.size),
   350→      onCollapseToolbar: (collapsed) => {
   351→        localStorage.setItem('society-lab-tb-collapsed', collapsed ? '1' : '0');
   352→      },
   353→    };
   354→  }

   368→  private blessingAt(s: Settlement) {
   369→    if (!this.world) return;
   370→    const res = castBrushPower(this.world, 'blessing', s.x, s.y, 0);
   371→    if (res.ok) {
   372→      const entry = powerChronicleEntry(this.world, 'blessing', res);
   373→      this.world.chronicle.push(entry);
   374→      this.vw?.addPing(s.x, s.y, 'rgba(232,196,98,0.95)');
   375→      this.audio.power();
   376→      this.flash('#e8c462', 0.3);
   377→      toast(`丰产祝福降临 ${s.name}`, 'good', 'blessing');
   378→      this.refreshHud();
   379→    } else toast(res.message, 'bad', 'alert');
   380→  }

   382→  private purifyAt(settlementId: number) {
   383→    if (!this.world) return false;
   384→    const res = castPurify(this.world, settlementId);
   385→    if (res.ok) {
   386→      this.world.chronicle.push({
   387→        id: `ev-${this.world.nextEventId++}`, year: this.world.year, title: '神力：净化疫病',
   388→        category: 'plague', text: '神光扫过病榻，高热退去，咳嗽止息。医者们把这一夜记录了下来。',
   389→        causes: ['神（玩家）净化疫病'], effects: ['疫病终止', '医学知识积累'], intervened: true,
   390→        severity: 0.4, settlementId,
   391→      });
   392→      const s = this.world.settlements.find(x => x.id === settlementId);
   393→      if (s) this.vw?.addPing(s.x, s.y, 'rgba(127,199,173,0.95)');
   394→      this.audio.power();
   395→      this.flash('#7fc7ad', 0.35);
   396→      toast('疫病已被净化', 'good', 'sparkle');
   397→      this.refreshHud();
   398→      return true;
   399→    }
   400→    toast(res.message, 'bad', 'alert');
   401→    return false;
   402→  }

   433→  private inputBound = false;
   434→  private bindCanvasInput(canvas: HTMLCanvasElement) {
   435→    if (this.inputBound) return;
   436→    this.inputBound = true;
   437→    let dragging = false;
   438→    let moved = 0;
   439→    let lastX = 0, lastY = 0;
   440→
   441→    canvas.addEventListener('mousedown', e => {
   442→      if (isModalOpen()) return;
   443→      dragging = true;
   444→      moved = 0;
   445→      lastX = e.clientX; lastY = e.clientY;
   446→    });
   447→    window.addEventListener('mousemove', e => {
   448→      if (!this.vw || !this.world) return;
   449→      // hover tile
   450→      const tile = this.vw.screenToTile(e.clientX, e.clientY);
   451→      this.vw.hoverTile = this.activePower && tile &&
   452→        tile.x >= 0 && tile.y >= 0 && tile.x < this.world.config.size && tile.y < this.world.config.size ? tile : null;
   453→      if (!dragging) return;
   454→      const dx = e.clientX - lastX, dy = e.clientY - lastY;
   455→      moved += Math.abs(dx) + Math.abs(dy);
   456→      if (moved > 6) {
   457→        this.vw.camera.x -= dx / this.vw.camera.zoom;
   458→        this.vw.camera.y -= dy / this.vw.camera.zoom;
   459→        this.vw.camera.targetX = this.vw.camera.x;
   460→        this.vw.camera.targetY = this.vw.camera.y;
   461→        canvas.classList.add('dragging');
   462→      }
   463→      lastX = e.clientX; lastY = e.clientY;
   464→    });
   465→    window.addEventListener('mouseup', e => {
   466→      if (!dragging) return;
   467→      dragging = false;
   468→      canvas.classList.remove('dragging');
   469→      if (moved <= 6 && e.target === canvas) this.handleClick(e.clientX, e.clientY, e.button);
   470→    });
   471→    canvas.addEventListener('wheel', e => {
   472→      if (!this.vw || !this.world) return;
   473→      e.preventDefault();
   474→      const factor = e.deltaY > 0 ? 0.88 : 1.14;
   475→      // 缩放下限随地图自适应：允许缩到全景之外一点
   476→      const rect = this.vw.canvas.getBoundingClientRect();
   477→      const fitZoom = Math.min(
   478→        rect.width / (this.world.config.size * 64),
   479→        rect.height / (this.world.config.size * 32 + 72),
   480→      ) * 0.96;
   481→      this.vw.camera.targetZoom = Math.max(fitZoom * 0.75, Math.min(2.8, this.vw.camera.targetZoom * factor));
   482→    }, { passive: false });
   483→    canvas.addEventListener('contextmenu', e => {
   484→      e.preventDefault();
   485→      if (this.activePower) {
   486→        this.activePower = null;
   487→        setHudPower(this.els.toolbar, null);
   488→        this.showHint('神力已取消', 1400);
   489→      }
   490→    });
   491→  }

   493→  private handleClick(px: number, py: number, button: number) {
   494→    if (!this.world || !this.vw || button !== 0) return;
   495→    const tile = this.vw.screenToTile(px, py);
   496→    if (!tile) return;
   497→    const size = this.world.config.size;
   498→    if (tile.x < 0 || tile.y < 0 || tile.x >= size || tile.y >= size) {
   499→      // 神力就绪时点到沙盘外给出提示，避免“点了没反应”的困惑（v1.0.1）
   500→      if (this.activePower) this.showHint('点击在沙盘之外 · 请点在地图上施放', 1800);
   501→      return;
   502→    }
   503→
   504→    if (this.activePower) {
   505→      const res = castBrushPower(this.world, this.activePower, tile.x, tile.y, this.brush);
   506→      if (res.ok) {
   507→        // 陨星在 castMeteor 内部已记录编年史，不重复记录
   508→        if (this.activePower !== 'meteor') {
   509→          const entry = powerChronicleEntry(this.world, this.activePower, res);
   510→          this.world.chronicle.push(entry);
   511→        }
   512→        // 地形变化 → 重建缓存与小地图
   513→        const terrainPowers: GodPowerId[] = ['forest', 'fertile', 'meadow', 'water', 'ore', 'uplift', 'meteor', 'wildfire'];
   514→        if (terrainPowers.includes(this.activePower)) {
   515→          this.vw.terrainCache.dirty = true;
   516→          this.vw.minimapDirty = true;
   517→        }
   518→        // 特效
   519→        this.vw.fx.spawnForPower(this.activePower, tile.x, tile.y, this.world);
   520→        this.vw.addPing(tile.x, tile.y, this.activePower === 'drought' || this.activePower === 'meteor' ? 'rgba(220,120,80,0.95)' : 'rgba(127,199,173,0.9)');
   521→        this.audio.power();
   522→        if (this.activePower === 'meteor') this.flash('#ffb060', 0.4);
   523→        else this.flash(this.activePower === 'drought' ? 'rgb(216,160,80)' : 'rgb(127,199,173)', 0.28);
   524→        toast(res.message, 'good', this.activePower);
   525→        this.refreshHud();
   526→      } else {
   527→        toast(res.message, 'bad', 'alert');
   528→      }
   529→      return;
   530→    }
   531→    // 检视模式：找最近聚落
   532→    let best: Settlement | null = null, bestD = 2.4;
   533→    for (const s of this.world.settlements) {
   534→      const d = Math.hypot(s.x - tile.x, s.y - tile.y);
   535→      if (d < bestD) { bestD = d; best = s; }
   536→    }
   537→    if (best) {
   538→      showSettlementPanel(this.els.sidepanel, this.world, best, this.panelActions());
   539→      this.audio.click();
   540→    } else {
   541→      hideSettlementPanel(this.els.sidepanel);
   542→    }
   543→  }

   546→  private bindKeys() {
   547→    if (this.keysBound) return;
   548→    this.keysBound = true;
   549→    document.addEventListener('keydown', e => {
   550→      if (isModalOpen()) {
   551→        if (e.key === 'Escape') {
   552→          // 必须走 closeModal：直接清 innerHTML 会让 modalOpen 标志永久卡在 true，
   553→          // 导致画布输入、快捷键与模拟推进全部失效（v1.0.1 修复）
   554→          closeModal();
   555→        }
   556→        return;
   557→      }
   558→      if (!this.world) return;
   559→      switch (e.key) {
   560→        case ' ':
   561→          e.preventDefault();
   562→          this.speed = this.speed === 0 ? 1 : 0;
   563→          setHudSpeed(this.els.topbar, this.speed);
   564→          break;
   565→        case '1': this.speed = 1; setHudSpeed(this.els.topbar, 1); break;
   566→        case '2': this.speed = 2; setHudSpeed(this.els.topbar, 2); break;
   567→        case '3': this.speed = 4; setHudSpeed(this.els.topbar, 4); break;
   568→        case '4': this.speed = 8; setHudSpeed(this.els.topbar, 8); break;
   569→        case 'Tab':
   570→          e.preventDefault();
   571→          {
   572→            const tb = this.els.toolbar;
   573→            const collapsed = !tb.classList.contains('collapsed');
   574→            setToolbarCollapsed(tb, collapsed);
   575→            localStorage.setItem('society-lab-tb-collapsed', collapsed ? '1' : '0');
   576→          }
   577→          break;
   578→        case 'Escape':
   579→          if (this.activePower) {
   580→            this.activePower = null;
   581→            setHudPower(this.els.toolbar, null);
   582→          } else {
   583→            if (!(this.els.sidepanel as HTMLElement).classList.contains('hidden')) collapseSettlementPanel(this.els.sidepanel);
   584→          }
   585→          break;
   586→        case 'h': case 'H':
   587→          this.vw?.fitMap(this.world.config.size);
   588→          break;
   589→        case 'c': case 'C': openChronicle(this.world, this.panelActions()); break;
   590→        case 'o': case 'O': openObserver(this.world); break;
   591→        case 's': case 'S': openSaveMenu(this.world, this.worldName, this.panelActions()); break;
   592→        case 'f': case 'F': {
   593→          // 聚焦最大聚落
   594→          const alive = this.world.settlements.filter(s => !s.dead);
   595→          if (alive.length) this.vw?.focusSettlement(alive.reduce((a, b) => a.population > b.population ? a : b));
   596→          break;
   597→        }
   598→      }
   599→    });
   600→    window.addEventListener('resize', () => {
   601→      this.vw?.resize();
   602→      if (this.menuVw) this.menuVw.resize();
   603→    });
   604→  }

   608→  frame(time: number) {
   609→    const dt = Math.min(0.06, (time - this.lastTime) / 1000 || 0.016);
   610→    this.lastTime = time;
   611→
   612→    // 菜单背景
   613→    if ((document.getElementById('menu') as HTMLElement).style.display !== 'none' && this.menuVw && this.menuWorld) {
   614→      const cam = this.menuVw.camera;
   615→      cam.targetX = Math.sin(time * 0.00008) * 380;
   616→      cam.targetY = this.menuWorld.config.size * 16 + Math.cos(time * 0.00006) * 160;
   617→      this.menuVw.render(this.menuWorld, dt, time, true);
   618→    }
   619→
   620→    // 游戏世界
   621→    if (this.world && this.vw && (document.getElementById('game') as HTMLElement).classList.contains('hidden') === false) {
   622→      if (this.speed > 0 && !isModalOpen()) {
   623→        this.yearAccum += dt * this.speed;
   624→        const yearSeconds = CFG.yearMs1x / 1000;
   625→        while (this.yearAccum >= yearSeconds) {
   626→          this.yearAccum -= yearSeconds;
   627→          const report = tickYear(this.world);
   628→          this.afterYear(report);
   629→          if (this.speed === 0) break;
   630→        }
   631→      }
   632→      this.vw.render(this.world, dt, time, this.speed === 0);
   633→    }
   634→    this.raf = requestAnimationFrame(t => this.frame(t));
   635→  }
   636→
   637→  afterYear(report: ReturnType<typeof tickYear>) {
   638→    if (!this.world) return;
   639→    const w = this.world;
   640→    this.refreshHud();
   641→
   642→    // 事件提示 + 视觉特效
   643→    for (const ev of report.events) {
   644→      if (ev.category === 'institution') {
   645→        this.playSfx('era');
   646→        toast(ev.title, 'good', 'branch');
   647→        const s = ev.settlementId !== undefined ? w.settlements.find(x => x.id === ev.settlementId) : null;
   648→        if (s) this.vw?.addPing(s.x, s.y, 'rgba(201,162,39,0.95)');
   649→      } else if (ev.severity > 0.55) {
   650→        this.playSfx('chime', ev.severity);
   651→        toast(ev.title, 'bad', 'alert');
   652→        const s = ev.settlementId !== undefined ? w.settlements.find(x => x.id === ev.settlementId) : null;
   653→        if (s) this.vw?.addPing(s.x, s.y, 'rgba(180,85,45,0.9)');
   654→      } else if (ev.category === 'invention' || ev.category === 'achievement') {
   655→        this.playSfx('milestone');
   656→      }
   657→      // 事件 → 沙盘上的真实视觉表现（v2.0）
   658→      this.vw?.fx.spawnForEvent(w, ev);
   659→    }
   660→
   661→    // 事件流
   662→    updateEventFeed(this.els.feed, w, this.panelActions(), this.lastFeedYear);
   663→    this.lastFeedYear = w.year;
   664→
   665→    // 雨声强度
   666→    if (this.vw && this.world) {
   667→      let wetCount = 0;
   668→      const step = 3;
   669→      for (let i = 0; i < this.world.weather.length; i += step * step) {
   670→        if (this.world.weather[i].wet > 0.5) wetCount++;
   671→      }
   672→      this.audio.setRainIntensity(Math.min(1, wetCount / 40));
   673→    }
   674→
   675→    // 自动存档（每 10 年）
   676→    if (w.year % 10 === 0) autosave(w, this.worldName);
   677→
   678→    // 文明湮灭提示
   679→    if (w.civ.vars.population === 0 && w.year > 10) {
   680→      this.speed = 0;
   681→      setHudSpeed(this.els.topbar, 0);
   682→      toast('文明已经湮灭……可读取存档或从分支重启', 'bad', 'alert');
   683→    }
   684→  }
