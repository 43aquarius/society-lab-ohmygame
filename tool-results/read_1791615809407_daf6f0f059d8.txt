    14→    14→const P = (wood: string, roof: string, stone = 'rgb(168,162,152)', accent = 'rgb(194,120,60)', cloth = 'rgb(224,200,150)'): BuildingPalette => ({
    15→    15→  wood, woodDark: shade(wood, 0.72), roof, roofDark: shade(roof, 0.7),
    16→    16→  stone, stoneDark: shade(stone, 0.72), accent, cloth,
    17→    17→});
    18→    18→
    19→    19→function shade(rgb: string, f: number): string {
    20→    20→  const m = rgb.match(/rgb\((\d+),(\d+),(\d+)\)/);
    21→    21→  if (!m) return rgb;
    22→    22→  return `rgb(${(+m[1]) * f | 0},${(+m[2]) * f | 0},${(+m[3]) * f | 0})`;
    23→    23→}

    33→    33→/** 建筑占地面积（格）：多数 1 格，大型 2 格 */
    34→    34→function footprint(kind: BuildingKind): { w: number; d: number } {
    35→    35→  switch (kind) {
    36→    36→    case BuildingKind.Admin: return { w: 1.6, d: 1.4 };
    37→    37→    case BuildingKind.Factory: return { w: 1.8, d: 1.6 };
    38→    38→    case BuildingKind.Services: return { w: 1.5, d: 1.3 };
    39→    39→    case BuildingKind.Cooperative: return { w: 1.4, d: 1.2 };
    40→    40→    case BuildingKind.Market: return { w: 1.5, d: 1.2 };
    41→    41→    case BuildingKind.Castle: return { w: 1.7, d: 1.5 };
    42→    42→    case BuildingKind.Manor: return { w: 1.6, d: 1.4 };
    43→    43→    case BuildingKind.Temple: return { w: 1.5, d: 1.3 };
    44→    44→    case BuildingKind.Academy: return { w: 1.3, d: 1.1 };
    45→    45→    case BuildingKind.Bank: return { w: 1.3, d: 1.1 };
    46→    46→    case BuildingKind.Theater: return { w: 1.4, d: 1.3 };
    47→    47→    case BuildingKind.Hospital: return { w: 1.2, d: 1.1 };
    48→    48→    case BuildingKind.Wall: return { w: 1.6, d: 0.7 };
    49→    49→    default: return { w: 1, d: 1 };
    50→    50→  }
    51→    51→}

    53→    53→/** 等距立方体：中心 (cx,cy)（地面），格尺寸 (gw, gd)，高 hp 像素 */
    54→    54→function box(ctx: CanvasRenderingContext2D, cx: number, cy: number, gw: number, gd: number, hp: number, top: string, left: string, right: string) {
    55→    55→  const hw = (gw * TILE_W) / 2, hd = (gd * TILE_H) / 2;
    56→    56→  // 左面
    57→    57→  ctx.fillStyle = left;
    58→    58→  ctx.beginPath();
    59→    59→  ctx.moveTo(cx - hw, cy);
    60→    60→  ctx.lineTo(cx, cy + hd);
    61→    61→  ctx.lineTo(cx, cy + hd - hp);
    62→    62→  ctx.lineTo(cx - hw, cy - hp);
    63→    63→  ctx.closePath();
    64→    64→  ctx.fill();
    65→    65→  // 右面
    66→    66→  ctx.fillStyle = right;
    67→    67→  ctx.beginPath();
    68→    68→  ctx.moveTo(cx + hw, cy);
    69→    69→  ctx.lineTo(cx, cy + hd);
    70→    70→  ctx.lineTo(cx, cy + hd - hp);
    71→    71→  ctx.lineTo(cx + hw, cy - hp);
    72→    72→  ctx.closePath();
    73→    73→  ctx.fill();
    74→    74→  // 顶面
    75→    75→  ctx.fillStyle = top;
    76→    76→  ctx.beginPath();
    77→    77→  ctx.moveTo(cx, cy - hp - hd);
    78→    78→  ctx.lineTo(cx + hw, cy - hp);
    79→    79→  ctx.lineTo(cx, cy - hp + hd);
    80→    80→  ctx.lineTo(cx - hw, cy - hp);
    81→    81→  ctx.closePath();
    82→    82→  ctx.fill();
    83→    83→  return { topY: cy - hp - hd, hw, hd };
    84→    84→}

    87→    87→function gableRoof(ctx: CanvasRenderingContext2D, cx: number, cy: number, gw: number, gd: number, hp: number, hrh: number, pal: BuildingPalette, flip = false) {
    88→    88→  const hw = (gw * TILE_W) / 2, hd = (gd * TILE_H) / 2;
    89→    89→  const peak = hp + hrh;
    90→    90→  if (!flip) {
    91→    91→    // 屋脊沿 SW-NE（左右两面）
    92→    92→    ctx.fillStyle = pal.roof;
    93→    93→    ctx.beginPath();
    94→    94→    ctx.moveTo(cx, cy + hd - peak);
    95→    95→    ctx.lineTo(cx + hw, cy - peak);
    96→    96→    ctx.lineTo(cx + hw, cy - hp);
    97→    97→    ctx.lineTo(cx, cy + hd - hp);
    98→    98→    ctx.closePath();
    99→    99→    ctx.fill();
   100→   100→    ctx.fillStyle = pal.roofDark;
   101→   101→    ctx.beginPath();
   102→   102→    ctx.moveTo(cx, cy + hd - peak);
   103→   103→    ctx.lineTo(cx - hw, cy - peak);
   104→   104→    ctx.lineTo(cx - hw, cy - hp);
   105→   105→    ctx.lineTo(cx, cy + hd - hp);
   106→   106→    ctx.closePath();
   107→   107→    ctx.fill();
   108→   108→  } else {
   109→   109→    ctx.fillStyle = pal.roof;
   110→   110→    ctx.beginPath();
   111→   111→    ctx.moveTo(cx, cy - hd - peak + hd);
   112→   112→    ctx.lineTo(cx + hw, cy - peak);
   113→   113→    ctx.lineTo(cx, cy + hd - peak + hd);
   114→   114→    ctx.closePath();
   115→   115→    ctx.fill();
   116→   116→  }
   117→   117→}

   119→   119→function shadow(ctx: CanvasRenderingContext2D, cx: number, cy: number, gw: number, gd: number) {
   120→   120→  const hw = (gw * TILE_W) / 2 + 3, hd = (gd * TILE_H) / 2 + 2;
   121→   121→  ctx.fillStyle = 'rgba(40,52,44,0.22)';
   122→   122→  ctx.beginPath();
   123→   123→  ctx.ellipse(cx + 2, cy + 2, hw * 0.8, hd * 0.8, 0, 0, Math.PI * 2);
   124→   124→  ctx.fill();
   125→   125→}

   127→   127→/** 烟囱烟雾 */
   128→   128→function smoke(ctx: CanvasRenderingContext2D, cx: number, cy: number, topY: number, time: number, seed: number, intensity = 1) {
   129→   129→  const n = 3;
   130→   130→  for (let i = 0; i < n; i++) {
   131→   131→    const ph = (time * 0.0004 + i / n + seed) % 1;
   132→   132→    const a = (1 - ph) * 0.28 * intensity;
   133→   133→    const r = 2.5 + ph * 7;
   134→   134→    ctx.fillStyle = `rgba(220,222,226,${a.toFixed(3)})`;
   135→   135→    ctx.beginPath();
   136→   136→    ctx.arc(cx + Math.sin(ph * 6 + seed * 9) * 4, topY - ph * 26, r, 0, Math.PI * 2);
   137→   137→    ctx.fill();
   138→   138→  }
   139→   139→}

   141→   141→/**
   142→   142→ * 绘制建筑。sx, sy 为建筑所在格的地面世界像素中心。
   143→   143→ * time 毫秒用于动画（炊烟等）。

   145→   145→export function drawBuilding(
   146→   146→  ctx: CanvasRenderingContext2D,
   147→   147→  w: WorldState,
   148→   148→  b: Building,
   149→   149→  sx: number, sy: number,
   150→   150→  time: number,
   151→   151→  lod: number,
   152→   152→) {
   153→   153→  const era = eraOf(w, b);
   154→   154→  const pal = PALETTES[era];
   155→   155→  const fp = footprint(b.kind);
   156→   156→  const seed = vhash(b.x, b.y, 5);
   157→   157→
   158→   158→  // 建造中：脚手架 + 半成品
   159→   159→  if (b.progress < 1) {
   160→   160→    shadow(ctx, sx, sy, fp.w, fp.d);
   161→   161→    ctx.globalAlpha = 0.35 + b.progress * 0.6;
   162→   162→    drawCompleted(ctx, w, b, sx, sy, time, era, pal, seed, lod, true);
   163→   163→    ctx.globalAlpha = 1;
   164→   164→    // 脚手架线条
   165→   165→    ctx.strokeStyle = 'rgba(120,90,56,0.8)';
   166→   166→    ctx.lineWidth = 1.5;
   167→   167→    const hw = (fp.w * TILE_W) / 2, hd = (fp.d * TILE_H) / 2;
   168→   168→    ctx.beginPath();
   169→   169→    ctx.moveTo(sx - hw, sy); ctx.lineTo(sx, sy + hd); ctx.lineTo(sx, sy + hd - 22); ctx.lineTo(sx - hw, sy - 22); ctx.closePath();
   170→   170→    ctx.moveTo(sx + hw, sy); ctx.lineTo(sx, sy + hd); ctx.lineTo(sx, sy + hd - 22); ctx.lineTo(sx + hw, sy - 22); ctx.closePath();
   171→   171→    ctx.stroke();
   172→   172→    return;
   173→   173→  }
   174→   174→  shadow(ctx, sx, sy, fp.w, fp.d);
   175→   175→  drawCompleted(ctx, w, b, sx, sy, time, era, pal, seed, lod, false);
   176→   176→}

   178→   178→function drawCompleted(
   179→   179→  ctx: CanvasRenderingContext2D,
   180→   180→  w: WorldState,
   181→   181→  b: Building,
   182→   182→  sx: number, sy: number,
   183→   183→  time: number,
   184→   184→  era: number,
   185→   185→  pal: BuildingPalette,
   186→   186→  seed: number,
   187→   187→  lod: number,
   188→   188→  ghost: boolean,
   189→   189→) {
   190→   190→  switch (b.kind) {
   191→   191→    case BuildingKind.Camp: {
   192→   192→      // 帐篷 + 篝火
   193→   193→      const h = 16;
   194→   194→      ctx.fillStyle = 'rgba(40,52,44,0.2)';
   195→   195→      ctx.beginPath(); ctx.ellipse(sx, sy + 2, 20, 9, 0, 0, Math.PI * 2); ctx.fill();
   196→   196→      ctx.fillStyle = pal.cloth;
   197→   197→      ctx.beginPath();
   198→   198→      ctx.moveTo(sx - 2, sy + 8);
   199→   199→      ctx.lineTo(sx - 16, sy - h);
   200→   200→      ctx.lineTo(sx + 2, sy - h - 2);
   201→   201→      ctx.closePath();
   202→   202→      ctx.fill();
   203→   203→      ctx.fillStyle = pal.roof;
   204→   204→      ctx.beginPath();
   205→   205→      ctx.moveTo(sx + 2, sy + 8);
   206→   206→      ctx.lineTo(sx + 14, sy - h + 2);
   207→   207→      ctx.lineTo(sx + 2, sy - h - 2);
   208→   208→      ctx.closePath();
   209→   209→      ctx.fill();
   210→   210→      ctx.fillStyle = 'rgba(60,42,26,0.85)';
   211→   211→      ctx.beginPath();
   212→   212→      ctx.moveTo(sx - 2, sy + 8); ctx.lineTo(sx - 9, sy - h + 8); ctx.lineTo(sx + 2, sy - h - 2);
   213→   213→      ctx.closePath(); ctx.fill();
   214→   214→      if (!ghost && lod > 0.5) {
   215→   215→        // 篝火
   216→   216→        const fl = 3 + Math.sin(time * 0.008 + seed * 9) * 1.4;
   217→   217→        ctx.fillStyle = 'rgb(224,120,50)';
   218→   218→        ctx.beginPath();
   219→   219→        ctx.ellipse(sx + 14, sy + 6, 3, fl, 0, 0, Math.PI * 2); ctx.fill();
   220→   220→        ctx.fillStyle = 'rgba(255,206,90,0.75)';
   221→   221→        ctx.beginPath();
   222→   222→        ctx.ellipse(sx + 14, sy + 5, 1.6, fl * 0.55, 0, 0, Math.PI * 2); ctx.fill();
   223→   223→      }
   224→   224→      break;
   225→   225→    }
   226→   226→    case BuildingKind.Shelter: {
   227→   227→      // 窝棚：树枝斜搭 + 茅草（v2.0 更精细：骨架线条 + 茅草层次）
   228→   228→      shadow(ctx, sx, sy, 0.7, 0.7);
   229→   229→      const h = 13;
   230→   230→      ctx.strokeStyle = pal.woodDark;
   231→   231→      ctx.lineWidth = 1.6;
   232→   232→      ctx.beginPath();
   233→   233→      ctx.moveTo(sx - 10, sy + 5); ctx.lineTo(sx - 2, sy - h);
   234→   234→      ctx.moveTo(sx + 10, sy + 5); ctx.lineTo(sx + 2, sy - h);
   235→   235→      ctx.moveTo(sx - 7, sy + 1); ctx.lineTo(sx + 1, sy - h + 3);
   236→   236→      ctx.stroke();
   237→   237→      // 茅草面（两片，带茅草簇）
   238→   238→      ctx.fillStyle = pal.roof;
   239→   239→      ctx.beginPath();
   240→   240→      ctx.moveTo(sx - 2, sy + 7); ctx.lineTo(sx - 13, sy - h + 2); ctx.lineTo(sx + 1, sy - h - 1);
   241→   241→      ctx.closePath(); ctx.fill();
   242→   242→      ctx.fillStyle = pal.roofDark;
   243→   243→      ctx.beginPath();
   244→   244→      ctx.moveTo(sx + 2, sy + 7); ctx.lineTo(sx + 12, sy - h + 3); ctx.lineTo(sx + 2, sy - h - 1);
   245→   245→      ctx.closePath(); ctx.fill();
   246→   246→      // 茅草纹理
   247→   247→      ctx.strokeStyle = 'rgba(120,96,52,0.5)';
   248→   248→      ctx.lineWidth = 0.7;
   249→   249→      ctx.beginPath();
   250→   250→      for (let i = 0; i < 3; i++) {
   251→   251→        ctx.moveTo(sx - 9 + i * 3, sy - h * 0.35 + i);
   252→   252→        ctx.lineTo(sx - 6 + i * 3, sy - h * 0.62 + i);
   253→   253→      }
   254→   254→      ctx.stroke();
   255→   255→      break;
   256→   256→    }
   257→   257→    case BuildingKind.House: {
   258→   258→      // 农舍 v2.0：木板墙纹理 + 双窗 + 门廊
   259→   259→      box(ctx, sx, sy, 0.85, 0.85, 14, pal.wood, pal.woodDark, shade(pal.wood, 0.85));
   260→   260→      // 木板横纹
   261→   261→      ctx.strokeStyle = 'rgba(90,64,42,0.35)';
   262→   262→      ctx.lineWidth = 0.8;
   263→   263→      ctx.beginPath();
   264→   264→      for (let i = 1; i <= 3; i++) {
   265→   265→        const yy = sy - 2 - i * 3;
   266→   266→        ctx.moveTo(sx - 13, yy - 13.5); ctx.lineTo(sx, yy - 19.5);
   267→   267→        ctx.moveTo(sx + 13, yy - 13.5); ctx.lineTo(sx, yy - 19.5);
   268→   268→      }
   269→   269→      ctx.stroke();
   270→   270→      gableRoof(ctx, sx, sy, 0.95, 0.95, 14, 8, pal);
   271→   271→      if (lod > 0.7) {
   272→   272→        ctx.fillStyle = 'rgba(255,214,140,0.85)';
   273→   273→        ctx.fillRect(sx + 4, sy - 8, 4, 4);
   274→   274→        ctx.fillRect(sx - 9, sy - 11, 3.5, 3.5);
   275→   275→        ctx.fillStyle = pal.woodDark;
   276→   276→        ctx.fillRect(sx - 5, sy - 5, 5, 6);
   277→   277→        ctx.fillStyle = 'rgba(40,30,20,0.5)';
   278→   278→        ctx.fillRect(sx - 5, sy - 5, 5, 1.2); // 门楣阴影
   279→   279→      }
   280→   280→      if (!ghost && lod > 0.6) smoke(ctx, sx - 6, sy, sy - 26, time, seed, 0.5);
   281→   281→      break;
   282→   282→    }
   283→   283→    case BuildingKind.Townhouse: {
   284→   284→      box(ctx, sx, sy, 0.8, 0.8, 30, pal.stone, pal.stoneDark, shade(pal.stone, 0.8));
   285→   285→      // 砖缝
   286→   286→      ctx.strokeStyle = 'rgba(90,88,84,0.3)';
   287→   287→      ctx.lineWidth = 0.7;
   288→   288→      ctx.beginPath();
   289→   289→      for (let i = 1; i <= 7; i++) {
   290→   290→        const yy = sy - i * 3.4;
   291→   291→        ctx.moveTo(sx - 12.5, yy - 12); ctx.lineTo(sx, yy - 17.5);
   292→   292→        ctx.moveTo(sx + 12.5, yy - 12); ctx.lineTo(sx, yy - 17.5);
   293→   293→      }
   294→   294→      ctx.stroke();
   295→   295→      gableRoof(ctx, sx, sy, 0.9, 0.9, 30, 9, pal);
   296→   296→      if (lod > 0.65) {
   297→   297→        for (let i = 0; i < 3; i++) {
   298→   298→          ctx.fillStyle = i % 2 ? 'rgba(255,214,140,0.85)' : 'rgba(150,170,180,0.8)';
   299→   299→          ctx.fillRect(sx + 4, sy - 6 - i * 9, 4, 4);
   300→   300→          ctx.fillRect(sx - 8, sy - 6 - i * 9, 4, 4);
   301→   301→        }
   302→   302→      }
   303→   303→      break;
   304→   304→    }
   305→   305→    case BuildingKind.Farm: {
   306→   306→      // 田垄与作物：随年内周期生长
   307→   307→      const growth = cropGrowth(w, b);
   308→   308→      drawField(ctx, sx, sy, growth, seed);
   309→   309→      break;
   310→   310→    }
   311→   311→    case BuildingKind.Granary: {
   312→   312→      // 架高粮仓
   313→   313→      ctx.strokeStyle = pal.woodDark;
   314→   314→      ctx.lineWidth = 2.5;
   315→   315→      ctx.beginPath();
   316→   316→      ctx.moveTo(sx - 12, sy + 4); ctx.lineTo(sx - 12, sy - 6);
   317→   317→      ctx.moveTo(sx + 12, sy + 4); ctx.lineTo(sx + 12, sy - 6);
   318→   318→      ctx.stroke();
   319→   319→      box(ctx, sx, sy - 6, 0.75, 0.75, 10, pal.wood, pal.woodDark, shade(pal.wood, 0.85));
   320→   320→      gableRoof(ctx, sx, sy - 6, 0.85, 0.85, 10, 7, pal);
   321→   321→      if (lod > 0.65) {
   322→   322→        ctx.fillStyle = 'rgb(224,196,120)';
   323→   323→        ctx.beginPath();
   324→   324→        ctx.ellipse(sx + 14, sy + 2, 3.5, 2.2, 0, 0, Math.PI * 2);
   325→   325→        ctx.fill();
   326→   326→      }
   327→   327→      break;
   328→   328→    }
   329→   329→    case BuildingKind.Well: {
   330→   330→      ctx.fillStyle = pal.stone;
   331→   331→      ctx.beginPath();
   332→   332→      ctx.ellipse(sx, sy, 9, 5, 0, 0, Math.PI * 2);
   333→   333→      ctx.fill();
   334→   334→      ctx.fillStyle = 'rgb(72,110,140)';
   335→   335→      ctx.beginPath();
   336→   336→      ctx.ellipse(sx, sy, 5.5, 3, 0, 0, Math.PI * 2);
   337→   337→      ctx.fill();
   338→   338→      ctx.strokeStyle = pal.woodDark;
   339→   339→      ctx.lineWidth = 2;
   340→   340→      ctx.beginPath();
   341→   341→      ctx.moveTo(sx - 7, sy); ctx.lineTo(sx - 4, sy - 14);
   342→   342→      ctx.moveTo(sx + 7, sy); ctx.lineTo(sx + 4, sy - 14);
   343→   343→      ctx.moveTo(sx - 5, sy - 13); ctx.lineTo(sx + 5, sy - 13);
   344→   344→      ctx.stroke();
   345→   345→      ctx.fillStyle = pal.roof;
   346→   346→      ctx.beginPath();
   347→   347→      ctx.moveTo(sx - 8, sy - 13); ctx.lineTo(sx, sy - 19); ctx.lineTo(sx + 8, sy - 13);
   348→   348→      ctx.closePath(); ctx.fill();
   349→   349→      break;
   350→   350→    }
   351→   351→    case BuildingKind.Workshop: {
   352→   352→      box(ctx, sx, sy, 0.9, 0.9, 16, pal.wood, pal.woodDark, shade(pal.wood, 0.82));
   353→   353→      gableRoof(ctx, sx, sy, 1.0, 1.0, 16, 8, pal);
   354→   354→      box(ctx, sx + 10, sy - 4, 0.22, 0.22, 22, pal.stone, pal.stoneDark, pal.stone);
   355→   355→      if (!ghost) smoke(ctx, sx + 10, sy - 4, sy - 30, time, seed);
   356→   356→      if (lod > 0.7) {
   357→   357→        ctx.fillStyle = pal.accent;
   358→   358→        ctx.fillRect(sx - 6, sy - 3, 4, 4); // 工件
   359→   359→      }
   360→   360→      break;
   361→   361→    }
   362→   362→    case BuildingKind.Market: {
   363→   363→      // 帐篷摊位

   756→   756→/** 农田：垄沟 + 随周期生长的作物色 */
   757→   757→function drawField(ctx: CanvasRenderingContext2D, sx: number, sy: number, growth: number, seed: number) {
   758→   758→  const hw = TILE_W / 2 - 4, hh = TILE_H / 2 - 2;
   759→   759→  // 土地
   760→   760→  ctx.fillStyle = 'rgb(146,110,72)';
   761→   761→  ctx.beginPath();
   762→   762→  ctx.moveTo(sx, sy - hh); ctx.lineTo(sx + hw, sy); ctx.lineTo(sx, sy + hh); ctx.lineTo(sx - hw, sy);
   763→   763→  ctx.closePath(); ctx.fill();
   764→   764→  // 垄
   765→   765→  ctx.strokeStyle = 'rgb(120,88,56)';
   766→   766→  ctx.lineWidth = 2;
   767→   767→  ctx.beginPath();
   768→   768→  for (let i = -2; i <= 2; i++) {
   769→   769→    const ox = i * hw / 2.6;
   770→   770→    ctx.moveTo(sx + ox, sy - hh / 2 + Math.abs(i) * 1.6);
   771→   771→    ctx.lineTo(sx + ox * 0.2, sy + hh / 2 - Math.abs(i) * 1.2);
   772→   772→  }
   773→   773→  ctx.stroke();
   774→   774→  // 作物
   775→   775→  if (growth > 0.15) {
   776→   776→    const g = Math.min(1, growth);
   777→   777→    const cropColor = g < 0.5 ? `rgb(${120 + g * 60},${150 + g * 50},${70})` : `rgb(${200 + g * 40},${168 - g * 40},${60})`;
   778→   778→    ctx.fillStyle = cropColor;
   779→   779→    for (let i = -2; i <= 2; i++) {
   780→   780→      for (let j = -1; j <= 1; j++) {
   781→   781→        const ox = i * hw / 3 + j * 4, oy = j * hh / 2.4 + i * 1.4;
   782→   782→        const hgt = 2 + g * 7;
   783→   783→        ctx.beginPath();
   784→   784→        ctx.ellipse(sx + ox, sy + oy - hgt * 0.4, 1.6, hgt * 0.55, 0, 0, Math.PI * 2);
   785→   785→        ctx.fill();
   786→   786→      }
   787→   787→    }
   788→   788→  }
   789→   789→}

   791→   791→function cropGrowth(w: WorldState, b: Building): number {
   792→   792→  // 以建造年 + 当前年的小数相位模拟周期生长（纯视觉）
   793→   793→  const phase = (w.year % 3) + ((b.id % 7) / 7);
   794→   794→  return 0.15 + 0.85 * Math.abs(Math.sin(phase * Math.PI / 3));
   795→   795→}

   797→   797→function eraOf(w: WorldState, b: Building): number {
   798→   798→  const t = w.civ.tech.levels;
   799→   799→  const lvl = (d: string) => t[d] ?? 1;
   800→   800→  switch (b.kind) {
   801→   801→    case BuildingKind.Camp: return 0;
   802→   802→    case BuildingKind.Shelter: return 0;
   803→   803→    case BuildingKind.Townhouse: case BuildingKind.Factory: return 3;
   804→   804→    case BuildingKind.Bank: case BuildingKind.Theater: return 3;
   805→   805→    case BuildingKind.Cooperative: case BuildingKind.Services: case BuildingKind.Hospital: return 4;
   806→   806→    case BuildingKind.Temple: case BuildingKind.Manor: case BuildingKind.Castle: return 2;
   807→   807→    case BuildingKind.Academy: return lvl('education') >= 4 ? 3 : 2;
   808→   808→    case BuildingKind.Market: case BuildingKind.Admin: case BuildingKind.Wall: case BuildingKind.Workshop: return 2;
   809→   809→    default: return lvl('agriculture') >= 2 ? 1 : 0;
   810→   810→  }
   811→   811→}
