   267→   267→  private hudCallbacks(): HudCallbacks {
   268→   268→    return {
   269→   269→      onSpeed: s => {
   270→   270→        this.speed = s;
   271→   271→        this.audio.ensure();
   272→   272→        setHudSpeed(this.els.topbar, s);
   273→   273→        if (this.world) this.showHint(s === 0 ? '时间暂停' : s === 1 ? '正常速度' : s === 2 ? '快速推进' : '极速推进', 1600);
   274→   274→      },
   275→   275→      onStep: () => {
   276→   276→        if (this.world && this.speed === 0) {
   277→   277→          const report = tickYear(this.world);
   278→   278→          this.afterYear(report);
   279→   279→        }
   280→   280→      },
   281→   281→      onPower: id => {
   282→   282→        if (!this.world) return;
   283→   283→        if (id === 'enlighten' || id === 'ideas') {
   284→   284→          const res = castCivPower(this.world, id);
   285→   285→          if (res.ok) {
   286→   286→            const entry = powerChronicleEntry(this.world, id, res);
   287→   287→            this.world.chronicle.push(entry);
   288→   288→            this.audio.power();
   289→   289→            this.flash('#7fc7ad', 0.35);
   290→   290→            toast(res.message, 'good', id === 'enlighten' ? 'enlighten' : 'ideas');
   291→   291→            this.refreshHud();
   292→   292→          } else toast(res.message, 'bad', 'alert');
   293→   293→          return;
   294→   294→        }
   295→   295→        this.activePower = this.activePower === id ? null : id;
   296→   296→        setHudPower(this.els.toolbar, this.activePower);
   297→   297→        this.showHint(this.activePower
   298→   298→          ? `神力就绪：点击地图施放（右键/Esc 取消）`
   299→   299→          : '检视模式：点击聚落查看详情', 2600);
   300→   300→      },
   301→   301→      onBrush: r => {
   302→   302→        this.brush = r;
   303→   303→        setHudBrush(this.els.toolbar, r);
   304→   304→        if (this.vw) this.vw.brushRadius = r;
   305→   305→      },
   306→   306→      onChronicle: () => this.world && openChronicle(this.world, this.panelActions()),
   307→   307→      onObserver: () => this.world && openObserver(this.world),
   308→   308→      onSave: () => this.world && openSaveMenu(this.world, this.worldName, this.panelActions()),
   309→   309→      onMenu: () => {
   310→   310→        if (this.world) { autosave(this.world, this.worldName); toast('已自动保存，返回主菜单', 'good', 'save'); }
   311→   311→        this.showMenu();
   312→   312→      },
   313→   313→      onHelp: () => openHelp(),
   314→   314→      onFormBadge: () => this.world && openFormCard(this.world),
   315→   315→      onPopStat: () => this.world && openObserver(this.world),
   316→   316→      onFoodStat: () => this.world && openObserver(this.world),
   317→   317→      onMute: () => {
   318→   318→        this.audio.setMuted(!this.audio.muted);
   319→   319→        setMuteIcon(this.els.topbar, this.audio.muted);
   320→   320→      },
   321→   321→      onHome: () => this.vw && this.world && this.vw.centerOnMap(this.world.config.size),
   322→   322→    };
   323→   323→  }

   337→   337→  private blessingAt(s: Settlement) {
   338→   338→    if (!this.world) return;

   340→   340→    if (res.ok) {

   342→   342→      this.world.chronicle.push(entry);
   343→   343→      this.vw?.addPing(s.x, s.y, 'rgba(232,196,98,0.95)');
   344→   344→      this.audio.power();
   345→   345→      this.flash('#e8c462', 0.3);

   347→   347→      this.refreshHud();
   348→   348→    } else toast(res.message, 'bad', 'alert');

   351→   351→  private purifyAt(settlementId: number) {
   352→   352→    if (!this.world) return false;
   353→   353→    const res = castPurify(this.world, settlementId);
   354→   354→    if (res.ok) {
   355→   355→      this.world.chronicle.push({
   356→   356→        id: `ev-${this.world.nextEventId++}`, year: this.world.year, title: '神力：净化疫病',
   357→   357→        category: 'plague', text: '神光扫过病榻，高热退去，咳嗽止息。医者们把这一夜记录了下来。',
   358→   358→        causes: ['神（玩家）净化疫病'], effects: ['疫病终止', '医学知识积累'], intervened: true,
   359→   359→        severity: 0.4, settlementId,
   360→   360→      });
   361→   361→      const s = this.world.settlements.find(x => x.id === settlementId);
   362→   362→      if (s) this.vw?.addPing(s.x, s.y, 'rgba(127,199,173,0.95)');
   363→   363→      this.audio.power();
   364→   364→      this.flash('#7fc7ad', 0.35);
   365→   365→      toast('疫病已被净化', 'good', 'sparkle');
   366→   366→      this.refreshHud();
   367→   367→      return true;
   368→   368→    }
   369→   369→    toast(res.message, 'bad', 'alert');
   370→   370→    return false;
   371→   371→  }

   411→   411→      if (isModalOpen()) return;

   421→   421→        tile.x >= 0 && tile.y >= 0 && tile.x < this.world.config.size && tile.y < this.world.config.size ? tile : null;
   422→   422→      if (!dragging) return;
   423→   423→      const dx = e.clientX - lastX, dy = e.clientY - lastY;
   424→   424→      moved += Math.abs(dx) + Math.abs(dy);
   425→   425→      if (moved > 6) {
   426→   426→        this.vw.camera.x -= dx / this.vw.camera.zoom;
   427→   427→        this.vw.camera.y -= dy / this.vw.camera.zoom;
   428→   428→        this.vw.camera.targetX = this.vw.camera.x;
   429→   429→        this.vw.camera.targetY = this.vw.camera.y;
   430→   430→        canvas.classList.add('dragging');
   431→   431→      }
   432→   432→      lastX = e.clientX; lastY = e.clientY;
   433→   433→    });
   434→   434→    window.addEventListener('mouseup', e => {
   435→   435→      if (!dragging) return;
   436→   436→      dragging = false;
   437→   437→      canvas.classList.remove('dragging');
   438→   438→      if (moved <= 6 && e.target === canvas) this.handleClick(e.clientX, e.clientY, e.button);
   439→   439→    });
   440→   440→    canvas.addEventListener('wheel', e => {
   441→   441→      if (!this.vw) return;
   442→   442→      e.preventDefault();
   443→   443→      const factor = e.deltaY > 0 ? 0.88 : 1.14;
   444→   444→      this.vw.camera.targetZoom = Math.max(0.4, Math.min(2.6, this.vw.camera.targetZoom * factor));
   445→   445→    }, { passive: false });
   446→   446→    canvas.addEventListener('contextmenu', e => {
   447→   447→      e.preventDefault();
   448→   448→      if (this.activePower) {
   449→   449→        this.activePower = null;
   450→   450→        setHudPower(this.els.toolbar, null);
   451→   451→        this.showHint('神力已取消', 1400);
   452→   452→      }
   453→   453→    });

   456→   456→  private handleClick(px: number, py: number, button: number) {
   457→   457→    if (!this.world || !this.vw || button !== 0) return;
   458→   458→    const tile = this.vw.screenToTile(px, py);
   459→   459→    if (!tile) return;
   460→   460→    const size = this.world.config.size;
   461→   461→    if (tile.x < 0 || tile.y < 0 || tile.x >= size || tile.y >= size) {
   462→   462→      // 神力就绪时点到沙盘外给出提示，避免“点了没反应”的困惑（v1.0.1）
   463→   463→      if (this.activePower) this.showHint('点击在沙盘之外 · 请点在地图上施放', 1800);
   464→   464→      return;
   465→   465→    }
   466→   466→
   467→   467→    if (this.activePower) {
   468→   468→      const res = castBrushPower(this.world, this.activePower, tile.x, tile.y, this.brush);
   469→   469→      if (res.ok) {
   470→   470→        const entry = powerChronicleEntry(this.world, this.activePower, res);
   471→   471→        this.world.chronicle.push(entry);
   472→   472→        this.vw.minimapDirty = true;
   473→   473→        this.vw.addPing(tile.x, tile.y, 'rgba(127,199,173,0.9)');
   474→   474→        this.audio.power();
   475→   475→        this.flash(this.activePower === 'drought' ? 'rgb(216,160,80)' : 'rgb(127,199,173)', 0.28);
   476→   476→        toast(res.message, 'good', this.activePower);
   477→   477→        this.refreshHud();
   478→   478→      } else {
   479→   479→        toast(res.message, 'bad', 'alert');
   480→   480→      }
   481→   481→      return;
   482→   482→    }
   483→   483→    // 检视模式：找最近聚落
   484→   484→    let best: Settlement | null = null, bestD = 2.4;
   485→   485→    for (const s of this.world.settlements) {
   486→   486→      const d = Math.hypot(s.x - tile.x, s.y - tile.y);
   487→   487→      if (d < bestD) { bestD = d; best = s; }
   488→   488→    }
   489→   489→    if (best) {
   490→   490→      showSettlementPanel(this.els.sidepanel, this.world, best, this.panelActions());
   491→   491→      this.audio.click();
   492→   492→    } else {
   493→   493→      hideSettlementPanel(this.els.sidepanel);
   494→   494→    }
   495→   495→  }

   550→   550→  frame(time: number) {
   551→   551→    const dt = Math.min(0.06, (time - this.lastTime) / 1000 || 0.016);
   552→   552→    this.lastTime = time;
   553→   553→
   554→   554→    // 菜单背景
   555→   555→    if ((document.getElementById('menu') as HTMLElement).style.display !== 'none' && this.menuVw && this.menuWorld) {
   556→   556→      const cam = this.menuVw.camera;
   557→   557→      cam.targetX = Math.sin(time * 0.00008) * 380;
   558→   558→      cam.targetY = this.menuWorld.config.size * 16 + Math.cos(time * 0.00006) * 160;
   559→   559→      this.menuVw.render(this.menuWorld, dt, time, true);
   560→   560→    }
   561→   561→
   562→   562→    // 游戏世界
   563→   563→    if (this.world && this.vw && (document.getElementById('game') as HTMLElement).classList.contains('hidden') === false) {
   564→   564→      if (this.speed > 0 && !isModalOpen()) {
   565→   565→        this.yearAccum += dt * this.speed;
   566→   566→        const yearSeconds = CFG.yearMs1x / 1000;
   567→   567→        while (this.yearAccum >= yearSeconds) {
   568→   568→          this.yearAccum -= yearSeconds;
   569→   569→          const report = tickYear(this.world);
   570→   570→          this.afterYear(report);
   571→   571→          if (this.speed === 0) break;
   572→   572→        }
   573→   573→      }
   574→   574→      this.vw.render(this.world, dt, time, this.speed === 0);
   575→   575→    }
   576→   576→    this.raf = requestAnimationFrame(t => this.frame(t));
   577→   577→  }
