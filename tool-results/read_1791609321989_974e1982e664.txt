   267→  private hudCallbacks(): HudCallbacks {
   268→    return {
   269→      onSpeed: s => {
   270→        this.speed = s;
   271→        this.audio.ensure();
   272→        setHudSpeed(this.els.topbar, s);
   273→        if (this.world) this.showHint(s === 0 ? '时间暂停' : s === 1 ? '正常速度' : s === 2 ? '快速推进' : '极速推进', 1600);
   274→      },
   275→      onStep: () => {
   276→        if (this.world && this.speed === 0) {
   277→          const report = tickYear(this.world);
   278→          this.afterYear(report);
   279→        }
   280→      },
   281→      onPower: id => {
   282→        if (!this.world) return;
   283→        if (id === 'enlighten' || id === 'ideas') {
   284→          const res = castCivPower(this.world, id);
   285→          if (res.ok) {
   286→            const entry = powerChronicleEntry(this.world, id, res);
   287→            this.world.chronicle.push(entry);
   288→            this.audio.power();
   289→            this.flash('#7fc7ad', 0.35);
   290→            toast(res.message, 'good', id === 'enlighten' ? 'enlighten' : 'ideas');
   291→            this.refreshHud();
   292→          } else toast(res.message, 'bad', 'alert');
   293→          return;
   294→        }
   295→        this.activePower = this.activePower === id ? null : id;
   296→        setHudPower(this.els.toolbar, this.activePower);
   297→        this.showHint(this.activePower
   298→          ? `神力就绪：点击地图施放（右键/Esc 取消）`
   299→          : '检视模式：点击聚落查看详情', 2600);
   300→      },
   301→      onBrush: r => {
   302→        this.brush = r;
   303→        setHudBrush(this.els.toolbar, r);
   304→        if (this.vw) this.vw.brushRadius = r;
   305→      },
   306→      onChronicle: () => this.world && openChronicle(this.world, this.panelActions()),
   307→      onObserver: () => this.world && openObserver(this.world),
   308→      onSave: () => this.world && openSaveMenu(this.world, this.worldName, this.panelActions()),
   309→      onMenu: () => {
   310→        if (this.world) { autosave(this.world, this.worldName); toast('已自动保存，返回主菜单', 'good', 'save'); }
   311→        this.showMenu();
   312→      },
   313→      onHelp: () => openHelp(),
   314→      onFormBadge: () => this.world && openFormCard(this.world),
   315→      onPopStat: () => this.world && openObserver(this.world),
   316→      onFoodStat: () => this.world && openObserver(this.world),
   317→      onMute: () => {
   318→        this.audio.setMuted(!this.audio.muted);
   319→        setMuteIcon(this.els.topbar, this.audio.muted);
   320→      },
   321→      onHome: () => this.vw && this.world && this.vw.centerOnMap(this.world.config.size),
   322→    };
   323→  }

   402→  private inputBound = false;
   403→  private bindCanvasInput(canvas: HTMLCanvasElement) {
   404→    if (this.inputBound) return;
   405→    this.inputBound = true;
   406→    let dragging = false;
   407→    let moved = 0;
   408→    let lastX = 0, lastY = 0;
   409→
   410→    canvas.addEventListener('mousedown', e => {
   411→      if (isModalOpen()) return;
   412→      dragging = true;
   413→      moved = 0;
   414→      lastX = e.clientX; lastY = e.clientY;
   415→    });
   416→    window.addEventListener('mousemove', e => {
   417→      if (!this.vw || !this.world) return;
   418→      // hover tile
   419→      const tile = this.vw.screenToTile(e.clientX, e.clientY);
   420→      this.vw.hoverTile = this.activePower && tile &&
   421→        tile.x >= 0 && tile.y >= 0 && tile.x < this.world.config.size && tile.y < this.world.config.size ? tile : null;
   422→      if (!dragging) return;
   423→      const dx = e.clientX - lastX, dy = e.clientY - lastY;
   424→      moved += Math.abs(dx) + Math.abs(dy);
   425→      if (moved > 6) {
   426→        this.vw.camera.x -= dx / this.vw.camera.zoom;
   427→        this.vw.camera.y -= dy / this.vw.camera.zoom;
   428→        this.vw.camera.targetX = this.vw.camera.x;
   429→        this.vw.camera.targetY = this.vw.camera.y;
   430→        canvas.classList.add('dragging');
   431→      }
   432→      lastX = e.clientX; lastY = e.clientY;
   433→    });
   434→    window.addEventListener('mouseup', e => {
   435→      if (!dragging) return;
   436→      dragging = false;
   437→      canvas.classList.remove('dragging');
   438→      if (moved <= 6 && e.target === canvas) this.handleClick(e.clientX, e.clientY, e.button);
   439→    });
   440→    canvas.addEventListener('wheel', e => {
   441→      if (!this.vw) return;
   442→      e.preventDefault();
   443→      const factor = e.deltaY > 0 ? 0.88 : 1.14;
   444→      this.vw.camera.targetZoom = Math.max(0.4, Math.min(2.6, this.vw.camera.targetZoom * factor));
   445→    }, { passive: false });
   446→    canvas.addEventListener('contextmenu', e => {
   447→      e.preventDefault();
   448→      if (this.activePower) {
   449→        this.activePower = null;
   450→        setHudPower(this.els.toolbar, null);
   451→        this.showHint('神力已取消', 1400);
   452→      }
   453→    });
   454→  }

   456→  private handleClick(px: number, py: number, button: number) {
   457→    if (!this.world || !this.vw || button !== 0) return;
   458→    const tile = this.vw.screenToTile(px, py);
   459→    if (!tile) return;
   460→    const size = this.world.config.size;
   461→    if (tile.x < 0 || tile.y < 0 || tile.x >= size || tile.y >= size) {
   462→      // 神力就绪时点到沙盘外给出提示，避免“点了没反应”的困惑（v1.0.1）
   463→      if (this.activePower) this.showHint('点击在沙盘之外 · 请点在地图上施放', 1800);
   464→      return;
   465→    }
   466→
   467→    if (this.activePower) {
   468→      const res = castBrushPower(this.world, this.activePower, tile.x, tile.y, this.brush);
   469→      if (res.ok) {
   470→        const entry = powerChronicleEntry(this.world, this.activePower, res);
   471→        this.world.chronicle.push(entry);
   472→        this.vw.minimapDirty = true;
   473→        this.vw.addPing(tile.x, tile.y, 'rgba(127,199,173,0.9)');
   474→        this.audio.power();
   475→        this.flash(this.activePower === 'drought' ? 'rgb(216,160,80)' : 'rgb(127,199,173)', 0.28);
   476→        toast(res.message, 'good', this.activePower);
   477→        this.refreshHud();
   478→      } else {
   479→        toast(res.message, 'bad', 'alert');
   480→      }
   481→      return;
   482→    }
   483→    // 检视模式：找最近聚落
   484→    let best: Settlement | null = null, bestD = 2.4;
   485→    for (const s of this.world.settlements) {
   486→      const d = Math.hypot(s.x - tile.x, s.y - tile.y);
   487→      if (d < bestD) { bestD = d; best = s; }
   488→    }
   489→    if (best) {
   490→      showSettlementPanel(this.els.sidepanel, this.world, best, this.panelActions());
   491→      this.audio.click();
   492→    } else {
   493→      hideSettlementPanel(this.els.sidepanel);
   494→    }
   495→  }

   497→  private keysBound = false;
   498→  private bindKeys() {
   499→    if (this.keysBound) return;
   500→    this.keysBound = true;
   501→    document.addEventListener('keydown', e => {
   502→      if (isModalOpen()) {
   503→        if (e.key === 'Escape') {
   504→          // 必须走 closeModal：直接清 innerHTML 会让 modalOpen 标志永久卡在 true，
   505→          // 导致画布输入、快捷键与模拟推进全部失效（v1.0.1 修复）
   506→          closeModal();
   507→        }
   508→        return;
   509→      }
   510→      if (!this.world) return;
   511→      switch (e.key) {
   512→        case ' ':
   513→          e.preventDefault();
   514→          this.speed = this.speed === 0 ? 1 : 0;
   515→          setHudSpeed(this.els.topbar, this.speed);
   516→          break;
   517→        case '1': this.speed = 1; setHudSpeed(this.els.topbar, 1); break;
   518→        case '2': this.speed = 2; setHudSpeed(this.els.topbar, 2); break;
   519→        case '3': this.speed = 4; setHudSpeed(this.els.topbar, 4); break;
   520→        case 'Escape':
   521→          if (this.activePower) {
   522→            this.activePower = null;
   523→            setHudPower(this.els.toolbar, null);
   524→          } else {
   525→            hideSettlementPanel(this.els.sidepanel);
   526→          }
   527→          break;
   528→        case 'h': case 'H':
   529→          this.vw?.centerOnMap(this.world.config.size);
   530→          break;
   531→        case 'c': case 'C': openChronicle(this.world, this.panelActions()); break;
   532→        case 'o': case 'O': openObserver(this.world); break;
   533→        case 's': case 'S': openSaveMenu(this.world, this.worldName, this.panelActions()); break;
   534→        case 'f': case 'F': {
   535→          // 聚焦最大聚落
   536→          const alive = this.world.settlements.filter(s => !s.dead);
   537→          if (alive.length) this.vw?.focusSettlement(alive.reduce((a, b) => a.population > b.population ? a : b));
   538→          break;
   539→        }
   540→      }
   541→    });
   542→    window.addEventListener('resize', () => {
   543→      this.vw?.resize();
   544→      if (this.menuVw) this.menuVw.resize();
   545→    });
   546→  }

   550→  frame(time: number) {
   551→    const dt = Math.min(0.06, (time - this.lastTime) / 1000 || 0.016);
   552→    this.lastTime = time;
   553→
   554→    // 菜单背景
   555→    if ((document.getElementById('menu') as HTMLElement).style.display !== 'none' && this.menuVw && this.menuWorld) {
   556→      const cam = this.menuVw.camera;
   557→      cam.targetX = Math.sin(time * 0.00008) * 380;
   558→      cam.targetY = this.menuWorld.config.size * 16 + Math.cos(time * 0.00006) * 160;
   559→      this.menuVw.render(this.menuWorld, dt, time, true);
   560→    }
   561→
   562→    // 游戏世界
   563→    if (this.world && this.vw && (document.getElementById('game') as HTMLElement).classList.contains('hidden') === false) {
   564→      if (this.speed > 0 && !isModalOpen()) {
   565→        this.yearAccum += dt * this.speed;
   566→        const yearSeconds = CFG.yearMs1x / 1000;
   567→        while (this.yearAccum >= yearSeconds) {
   568→          this.yearAccum -= yearSeconds;
   569→          const report = tickYear(this.world);
   570→          this.afterYear(report);
   571→          if (this.speed === 0) break;
   572→        }
   573→      }
   574→      this.vw.render(this.world, dt, time, this.speed === 0);
   575→    }
   576→    this.raf = requestAnimationFrame(t => this.frame(t));
   577→  }
   578→
   579→  afterYear(report: ReturnType<typeof tickYear>) {
   580→    if (!this.world) return;
   581→    const w = this.world;
   582→    this.refreshHud();
   583→
   584→    // 事件提示
   585→    for (const ev of report.events) {
   586→      if (ev.category === 'institution') {
   587→        this.playSfx('era');
   588→        toast(ev.title, 'good', 'branch');
   589→        const s = ev.settlementId !== undefined ? w.settlements.find(x => x.id === ev.settlementId) : null;
   590→        if (s) this.vw?.addPing(s.x, s.y, 'rgba(201,162,39,0.95)');
   591→      } else if (ev.severity > 0.55) {
   592→        this.playSfx('chime', ev.severity);
   593→        toast(ev.title, 'bad', 'alert');
   594→        const s = ev.settlementId !== undefined ? w.settlements.find(x => x.id === ev.settlementId) : null;
   595→        if (s) this.vw?.addPing(s.x, s.y, 'rgba(180,85,45,0.9)');
   596→      } else if (ev.category === 'invention' || ev.category === 'achievement') {
   597→        this.playSfx('milestone');
   598→      }
   599→      if (ev.settlementId !== undefined) {
   600→        const s = w.settlements.find(x => x.id === ev.settlementId);
   601→        if (s && ev.category !== 'institution' && ev.severity > 0.55) this.vw?.addPing(s.x, s.y, 'rgba(220,120,80,0.9)');
   602→      }
   603→    }
   604→
   605→    // 事件流
   606→    updateEventFeed(this.els.feed, w, this.panelActions(), this.lastFeedYear);
   607→    this.lastFeedYear = w.year;
   608→
   609→    // 雨声强度
   610→    if (this.vw && this.world) {
   611→      let wetCount = 0;
   612→      const step = 3;
   613→      for (let i = 0; i < this.world.weather.length; i += step * step) {
   614→        if (this.world.weather[i].wet > 0.5) wetCount++;
   615→      }
   616→      this.audio.setRainIntensity(Math.min(1, wetCount / 40));
   617→    }
   618→
   619→    // 自动存档（每 10 年）
   620→    if (w.year % 10 === 0) autosave(w, this.worldName);
   621→
   622→    // 文明湮灭提示
   623→    if (w.civ.vars.population === 0 && w.year > 10) {
   624→      this.speed = 0;
   625→      setHudSpeed(this.els.topbar, 0);
   626→      toast('文明已经湮灭……可读取存档或从分支重启', 'bad', 'alert');
   627→    }
   628→  }
